#Santos 412: Crossing: Wind Shear
santos.412 = {
	dlc = dlc_morgenrote

	type = country_event
	hidden = yes

	cooldown = {
		days = 14
	}

	trigger = {
		OR = {
			var:santos_balloon_crossing_target_var ?= flag:santos_bab_flag
			var:santos_balloon_crossing_target_var ?= flag:santos_pyrenees_flag
			var:santos_balloon_crossing_target_var ?= flag:santos_alps_flag
			var:santos_balloon_crossing_target_var ?= flag:santos_andes_flag
		}
	}

	immediate = {
		santos_balloon_crossing_log_wind_shear_event_effect = yes
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:santos_balloonist_var ?= {
				has_trait = ambitious
			}
		  	factor = 3 #Ambitious Balloonist the best to do this
		}
		modifier = {
			var:santos_balloonist_var ?= {
				has_trait = santos_balloonist_hero
			}
		  	factor = 2 #Also quite good
		}
		modifier = {
			var:santos_balloonist_var.var:santos_balloonist_experience_var >= 150
		  	factor = 1.5 #Experienced Balloonist will try
		}
	}
}
