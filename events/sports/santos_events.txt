#Santos 601: Balloonist tries to set new Altitude record (hidden)
santos.601 = {
	dlc = dlc_morgenrote

	type = country_event
	hidden = yes

	cooldown = {
		days = 365 #Should not do two attempts in a row
	}

	trigger = {
		#triggered by on_action in journal entry
		has_variable = santos_balloonist_var
	}

	immediate = {
		santos_new_balloon_altitude_effect = yes

		if = {
			limit = {
				santos_altitude_record_global_broken_trigger = yes
			}
			medium_santos_ballooning_research_effect = yes
			santos_balloon_country_altitude_record_effect = yes
		}
		else_if = {
			limit = {
				santos_altitude_record_country_broken_trigger = yes
				santos_altitude_record_global_broken_trigger = no
			}
			small_santos_ballooning_research_effect = yes
			santos_balloon_country_altitude_record_effect = yes
		}
		else = {
			santos_balloon_country_altitude_record_effect = yes
		}

		balloonist_level_up_effect = yes
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:santos_balloonist_var ?= {
				var:santos_balloon_type_var ?= flag:santos_balloon_type_altitude_flag
			}
			factor = 1.5 #Altitude Balloons will try to do altitude record
		}
		modifier = {
			var:santos_balloonist_var ?= {
				has_trait = innovative
			}
		  	factor = 1.5 #Innovative balloonist will try to get high (because he's a space cowboy and tries to get by)
		}
	}
}

#Santos 652: Balloonist's first flight
santos.652 = {
	dlc = dlc_morgenrote

	type = country_event

	title = santos.652.t

	desc = santos.652.d

	flavor = santos.652.f

	duration = 2

	gui_window = event_window_1char_tabloid

	left_icon = scope:santos_balloonist_scope

	icon = "gfx/interface/icons/event_icons/event_ballooning.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/airship"

	trigger = {
		has_variable = santos_balloonist_var
		OR = {
			trigger_if = {
				limit = { has_variable = santos_balloon_altitude_record_var }
				var:santos_balloon_altitude_record_var = 0
			}
			trigger_if = {
				limit = { has_variable = santos_balloon_distance_record_var }
				var:santos_balloon_distance_record_var = 0
			}
			trigger_if = {
				limit = { has_variable = santos_balloon_time_record_var }
				var:santos_balloon_time_record_var = 0
			}
		}
		NOT = { #only happens if player wants to see results.
			has_variable = santos_no_balloonist_flight_events
		}
	}

	cancellation_trigger = {
		NOR = {
			has_variable = santos_balloonist_var
			exists = var:santos_balloonist_var
		}
	}


	immediate = {
		var:santos_balloonist_var = {
			save_scope_as = santos_balloonist_scope
		}

		santos_new_balloon_altitude_effect = yes
		santos_new_balloon_distance_effect = yes
		santos_new_balloon_time_effect = yes
	}

	option = {	#That's a good start!
		name = santos.652.g
		default_option = yes

		santos_balloon_country_altitude_experience_only_world_record_effect = yes
		santos_balloon_country_distance_record_experience_only_world_record_effect = yes
		santos_balloon_country_time_record_experience_only_world_record_effect = yes

		santos_ballooning_research_if_record_all_effect = yes

		balloonist_level_up_effect = yes
	}

	option = {	#I don't want to hear about the flights of my balloonist anymore!
		name = santos.651.w

		trigger = {
			is_player = yes
		}

		custom_tooltip = santos.651.w.tt

		set_variable = santos_no_balloonist_flight_events

		santos_balloon_country_altitude_experience_only_world_record_effect = yes
		santos_balloon_country_distance_record_experience_only_world_record_effect = yes
		santos_balloon_country_time_record_experience_only_world_record_effect = yes

		santos_ballooning_research_if_record_all_effect = yes

		balloonist_level_up_effect = yes

	}
}
