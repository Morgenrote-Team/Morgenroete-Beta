#Curtiss 1001: Introducing our air ship captain
zeppelin.101 = {
	dlc = dlc_morgenrote

	type = country_event

	title = zeppelin.101.t
	desc = zeppelin.101.d
	flavor = zeppelin.101.f

	duration = 2

	gui_window = event_window_1char_tabloid

	left_icon = scope:zeppelin_new_captain_scope

	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		#triggered by decision
	}

	immediate = {
		create_new_character_base_effect = { CHARACTER = captain PROJECT = zeppelin }

		var:zeppelin_new_captain_var = {
			save_scope_as = zeppelin_new_captain_scope
		}
		zeppelin_set_new_captain_as_captain_effect = yes
	}

	option = {	#Good Luck!
		name = zeppelin.101.g
		default_option = yes
		show_as_tooltip = {
			zeppelin_set_new_captain_as_captain_effect = yes

#Curtiss 1002: Airship Captain becomes Admiral of the Skies
zeppelin.102 = {
	dlc = dlc_morgenrote

	type = country_event

	title = zeppelin.102.t
	desc = zeppelin.102.d
	flavor = zeppelin.102.f

	event_image = {
		texture = "gfx/event_pictures/curtiss_pictures/curtiss_zeppelin.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	duration = 3

	trigger = {
	}

	immediate = {
		random_scope_character = {
			limit = {
				has_role_of_type = ruler
			}
			save_scope_as = zeppelin_ruler_scope
		}
	}

	option = { #A new national hero!
		name = zeppelin.102.g
		default_option = yes
		scope:curtiss_ace = {
			remove_trait = zeppelin_captain
			add_trait = zeppelin_captain_admiral
		}
	}
}
