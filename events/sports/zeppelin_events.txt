#Zeppelin 101: Introducing our air ship captain
zeppelin.101 = {
	dlc = dlc_morgenrote

	type = country_event

	title = zeppelin.101.t

	desc = {
		triggered_desc = {
			trigger = {
				navy_size >= 1 #At least one ship for captain
			}
			desc = zeppelin.101.d
		}
		triggered_desc = {
			trigger = {
				navy_size = 0 #Landlocked nation
			}
			desc = zeppelin.101.d_no_ship
		}
	}

	flavor = {
		triggered_desc = {
			trigger = {
				navy_size >= 1 #At least one ship for captain
			}
			desc = zeppelin.101.f
		}
		triggered_desc = {
			trigger = {
				navy_size = 0 #Landlocked nation
			}

		}

	gui_window = event_window_1char_tabloid

		#triggered by decision

		name = zeppelin.101.g
		default_option = yes
		show_as_tooltip = {
			zeppelin_set_new_captain_as_captain_effect = yes

#Zeppelin 131: Maiden Voyage
zeppelin.131 = {
	dlc = dlc_morgenrote

	placement = scope:zeppelin_state_scope

	type = country_event

	title = zeppelin.131.t

	desc = {
		triggered_desc = {
			trigger = {
				var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_small_flag }
			}
			desc = zeppelin.131.d_small
		}
		triggered_desc = {
			trigger = {
				var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_medium_flag }
			}
			desc = zeppelin.131.d_medium
		}
		triggered_desc = {
			trigger = {
				var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_large_flag }
			}
			desc = zeppelin.131.d_large
		}
		triggered_desc = {
			trigger = {
				var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_giant_flag }
			}
			desc = zeppelin.131.d_giant
		}
	}

	flavor = zeppelin.131.f

	event_image = { #Small
		trigger = {
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_small_flag }
		}
		texture = "gfx/event_pictures/zeppelin_pictures/zeppelin_small.dds"
	}

	event_image = { #Medium
		trigger = {
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_medium_flag }
		}
		texture = "gfx/event_pictures/zeppelin_pictures/zeppelin_medium.dds"
	}

	event_image = { #Large
		trigger = {
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_large_flag }
		}
		texture = "gfx/event_pictures/zeppelin_pictures/zeppelin_large.dds"
	}

	event_image = { #Giant
		trigger = {
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_giant_flag }
		}
		texture = "gfx/event_pictures/zeppelin_pictures/zeppelin_giant.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	duration = 3

	trigger = {
	}

	immediate = {
		random_scope_character = {
			limit = {
				has_role_of_type = ruler
			}
			save_scope_as = zeppelin_ruler_scope
		}
		if = {
			limit = {
				any_scope_state = {
					any_scope_building = {
						is_building_type = building_zeppelin_hangar
					}
				}
			}
			random_scope_state = {
				limit = {
					any_scope_building = {
						is_building_type = building_zeppelin_hangar
					}
				}
				save_scope_as = zeppelin_state_scope
			}
		}
		else = {
			random_scope_state = {
				limit = {
					is_capital = yes
				}
				save_scope_as = zeppelin_state_scope
			}
		}
	}

	option = {	#I salute you!
		name = zeppelin.131.g
		if = {
			limit = {
				zeppelin_size_giant_trigger = yes
			}
			captain_experience_gain_effect = { AMOUNT = 10 }
			scope:zeppelin_state_scope = {
				add_loyalists_in_state = {
					value = 0.1
				}
			}
			add_modifier = {
				name = zeppelin_christening_modifier
				days = 365
			}
		}
		else_if = {
			limit = {
				zeppelin_size_large_trigger = yes
			}
			captain_experience_gain_effect = { AMOUNT = 5 }
			scope:zeppelin_state_scope = {
				add_loyalists_in_state = {
					value = 0.05
				}
			}
		}
	}
}

#Zeppelin 202: Expedition: Return
zeppelin.202 = {
	dlc = dlc_morgenrote

	type = country_event

	title = zeppelin.202.t
	desc = zeppelin.202.d
	flavor = zeppelin.202.f

	event_image = {
		texture = "gfx/event_pictures/zeppelin_pictures/zeppelin_arctic.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	duration = 3

	trigger = {
	}

	immediate = {
	}

	option = { #Glory
		name = zeppelin.202.g
		default_option = yes
		#Captain Experience
		captain_experience_gain_effect = { AMOUNT = 15 }
		#Arctic Map Bonus
		if = { 
			limit = { 
				var:zeppelin_captain_var ?= { 
					var:zeppelin_slot_1_var ?= flag:zeppelin_slot_1_station_flag 
				} 
			} 
			zeppelin_random_arctic_region_effect = yes
		}
		#Physics Bonus
		if = {
			limit = {
				has_variable = zeppelin_expedition_physics_var
				var:zeppelin_captain_var ?= { var:zeppelin_slot_3_var ?= flag:zeppelin_slot_3_magnetometer_flag }
			}
			curie_add_x_physical_data_effect = { AMOUNT = 60 }
			curie_add_x_magnetic_data_effect = { AMOUNT = 40 }
		}
		else_if = {
			limit = {
				OR = {
					has_variable = zeppelin_expedition_physics_var
					var:zeppelin_captain_var ?= { var:zeppelin_slot_3_var ?= flag:zeppelin_slot_3_magnetometer_flag }
				}
			}
			curie_add_x_physical_data_effect = { AMOUNT = 30 }
			curie_add_x_magnetic_data_effect = { AMOUNT = 20 }
		}
		else = {
			curie_add_x_physical_data_effect = { AMOUNT = 15 }
			curie_add_x_magnetic_data_effect = { AMOUNT = 10 }
		}
		#Nature Bonus
		if = {
			limit = {
				has_variable = zeppelin_expedition_nature_var
				var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_cage_flag }
			}
			dubois_add_x_specimen_effect = { AMOUNT = 6000 }
			agassiz_add_x_generic_minerals_effect = { AMOUNT = 4000 }
			dubois_add_x_fossils_effect = { AMOUNT = 4 }
		}
		else_if = {
			limit = {
				OR = {
					has_variable = zeppelin_expedition_nature_var
					var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_cage_flag }
				}
			}
			dubois_add_x_specimen_effect = { AMOUNT = 3000 }
			agassiz_add_x_generic_minerals_effect = { AMOUNT = 2000 }
			dubois_add_x_fossils_effect = { AMOUNT = 2 }
		}
		else = {
			dubois_add_x_specimen_effect = { AMOUNT = 1500 }
			agassiz_add_x_generic_minerals_effect = { AMOUNT = 1000 }
			dubois_add_x_fossils_effect = { AMOUNT = 1 }
		}
		add_modifier = {
			name = zeppelin_recent_expedition_success_modifier
			years = 1
		}
	}

	after = {
		zeppelin_finish_expedition_effect = yes
	}
}

zeppelin.212 = {
	dlc = dlc_morgenrote

	type = country_event
	hidden = yes

	cooldown = {
		days = 28
	}

	trigger = {
		je:je_zeppelin_expedition ?= {
			"scripted_bar_progress(zeppelin_expedition_danger_bar)" >= 1
		}
	}

	immediate = {
		zeppelin_expedition_log_zeppelin_anchorage_event_effect = yes
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:zeppelin_captain_var ?= {
				has_trait = cautious
			}
		  	factor = 2 #Cautious Captain will try to fly more secure after risk has grown
		}
		modifier = {
			var:zeppelin_captain_var ?= {
