#Curie 12: Joachimsthal
curie.12 = {
	dlc = dlc_morgenrote

	type = country_event

	title = curie.12.t

	desc = curie.12.d

	flavor = curie.10.f
	
	duration = 3

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		NOT = {
			has_global_variable = curie_joachimsthal_global_var
		}
		has_technology_researched = verrier_radioactivity_tech
		any_scope_state = {
			state_region = s:STATE_BOHEMIA
		}
	}

	immediate = {
		set_global_variable = curie_joachimsthal_global_var
		random_scope_state = {
			limit = {
				state_region = s:STATE_BOHEMIA
			}
			save_scope_as = curie_uranium_state_scope
		}
	}

	option = {	#Build a uranium mine right there!
		name = curie.12.g
		default_option = yes
		scope:curie_uranium_state_scope = {
			create_building = {
				building = building_uranium_mine
				level = 1
			}
		}
	}
}

	flavor = curie.100.f

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	trigger = {
	}

	immediate = {
		if = {
			limit = { has_variable = curie_physicist_var }
			var:curie_physicist_var = {
				set_variable = {
					name = curie_physicist_achievements_foundes_magnetic_association_var
					value = flag:curie_physicist_achievements_founded_geomagnetic_association_loc
				}
			}
		}
	}

	option = {
		name = curie.102.g #We should build a measuring station immediately.
		default_option = yes
		if = {
			limit = {
				curie_has_mathematical_physicist_trigger = yes
			}
			mathematical_physicist_experience_gain_effect = {
				AMOUNT = 30
			}
		}
		if = {
			limit = {
				curie_has_meteorological_physicist_trigger = yes
			}
			mathematical_physicist_experience_gain_effect = {
				AMOUNT = 30
			}
		}
		curie_add_x_magnetic_data = {
			AMOUNT = 5
		}
		curie_add_x_physical_data_effect = {
			AMOUNT = 2
		}
		random_scope_state = {
			limit = {
				has_building = building_verrier_observatory
			}
			activate_production_method = {
				building_type = building_verrier_observatory
				production_method = pm_curie_magnetic_observatory
			}
		}
	}
}

				is_busy = no
			}
			save_scope_as = curie_admiral_scope
		}
		var:curie_physicist_var = {
			save_scope_as = curie_physicist_scope
		}
	}

		name = curie.104.g #The new knowledge should be useful to him.
		default_option = yes
		scope:curie_admiral_scope = {
			add_modifier = {
				name = curie_geomagnetical_knowledge_modifier

		}

		name = curie.104.h #This knowledge should benefit everyone.
		curie_add_x_physical_data_effect = {
			AMOUNT = 2
		}
