#Curie 12: Joachimsthal
curie.12 = {
	dlc = dlc_morgenrote

	type = country_event

	title = curie.12.t

	desc = curie.12.d

	flavor = curie.10.f
	
	duration = 3

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		NOT = {
			has_global_variable = curie_joachimsthal_global_var
		}
		has_technology_researched = verrier_radioactivity_tech
		any_scope_state = {
			state_region = s:STATE_BOHEMIA
		}
	}

	immediate = {
		set_global_variable = curie_joachimsthal_global_var
		random_scope_state = {
			limit = {
				state_region = s:STATE_BOHEMIA
			}
			save_scope_as = curie_uranium_state_scope
		}
	}

	option = {	#Build a uranium mine right there!
		name = curie.12.g
		default_option = yes
		scope:curie_uranium_state_scope = {
			create_building = {
				building = building_uranium_mine
				level = 1
			}
		}
	}
}

#Gaudí 1401: Admission to the Göttingen Magnetic Association
gaudi.1401 = {
	dlc = dlc_morgenrote

	type = country_event

	title = gaudi.1401.t
	desc = gaudi.1401.d
	flavor = gaudi.1401.f

	placement = scope:gaudi_association_state

	gui_window = event_window_1char_tabloid

	left_icon = scope:gaudi_engineer_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/electrical_generation.dds"

	trigger = {
	}

	immediate = {
		var:tesla_engineer_var = {
			save_scope_as = gaudi_engineer_scope
		}
		random_state = {
			limit = {
				state_region = s:STATE_HANNOVER
			}
			save_scope_as = gaudi_association_state
			owner = {
				save_scope_as = gaudi_goettingen_owner
			}
		}
	}

	option = {
		name = gaudi.1401.g #We should build a measuring station immediately.
		default_option = yes
		scope:gaudi_engineer_scope = {
			set_character_busy_and_immortal = yes
		}
		electrical_engineer_experience_gain_effect = { AMOUNT = 15 }
		curie_add_magnetic_association_member = yes
		tesla_electrical_engineering_small_add_technology = yes
		every_country = {
			limit = {
				has_variable = curie_magnetic_association_member_var
				NOT = {
					ROOT = THIS
				}
			}
			change_relations = {
				country = ROOT
				value = 2
			}
		}
	}
}

		}
		curie_add_x_magnetic_data = {
			AMOUNT = 5
		}
		random_scope_state = {
			limit = {
				has_building = building_verrier_observatory
			}
			activate_production_method = {
				building_type = building_verrier_observatory

		years = 5
	}

		any_scope_character = {
			has_role_of_type = admiral
			is_busy = no
		}
	}

	immediate = {
		random_scope_character = {
			limit = {
				has_role_of_type = admiral
				is_busy = no
			}
			save_scope_as = gaudi_admiral_scope
		}
		var:tesla_engineer_var = {
			save_scope_as = gaudi_engineer_scope
		}
	}

	option = {
		name = gaudi.1404.g #The new knowledge should be useful to him.
		default_option = yes
		scope:gaudi_admiral_scope = {
			add_modifier = {
				name = gaudi_geomagnetical_knowledge_modifier
			}
		}
	}

	option = {
		name = gaudi.1404.h #This knowledge should benefit everyone.
		tesla_electrical_engineering_small_add_technology = yes
		curie_add_x_magnetic_data = {
			AMOUNT = 1
		}
	}
}
