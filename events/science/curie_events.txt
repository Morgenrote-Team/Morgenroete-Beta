#Curie 12: Joachimsthal
curie.12 = {
	dlc = dlc_morgenrote

	type = country_event

	title = curie.12.t

	desc = curie.12.d

	flavor = curie.10.f
	
	duration = 3

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		NOT = {
			has_global_variable = curie_joachimsthal_global_var
		}
		has_technology_researched = verrier_radioactivity_tech
		any_scope_state = {
			state_region = s:STATE_BOHEMIA
		}
	}

	immediate = {
		set_global_variable = curie_joachimsthal_global_var
		random_scope_state = {
			limit = {
				state_region = s:STATE_BOHEMIA
			}
			save_scope_as = curie_uranium_state_scope
		}
	}

	option = {	#Build a uranium mine right there!
		name = curie.12.g
		default_option = yes
		scope:curie_uranium_state_scope = {
			create_building = {
				building = building_uranium_mine
				level = 1
			}
		}
	}
}

#Curie 101: Admission to the Göttingen Magnetic Association
curie.101 = {
	dlc = dlc_morgenrote

	type = country_event

	title = curie.101.t
	desc = curie.101.d
	flavor = curie.101.f

	placement = scope:curie_association_state

	gui_window = event_window_1char_tabloid

	left_icon = scope:curie_physicist_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	trigger = {
	}

	immediate = {
		var:curie_physicist_var = {
			save_scope_as = curie_physicist_scope
		}
		random_state = {
			limit = {
				state_region = s:STATE_HANNOVER
			}
			save_scope_as = curie_association_state
			owner = {
				save_scope_as = curie_goettingen_owner
			}
		}
	}

	option = {
		name = curie.101.g #We should build a measuring station immediately.
		default_option = yes
		scope:curie_physicist_scope = {
			set_character_busy_and_immortal = yes
		}
		mathematical_physicist_experience_gain_effect = { AMOUNT = 15 }
		curie_add_magnetic_association_member = yes
		every_country = {
			limit = {
				has_variable = curie_magnetic_association_member_var
				NOT = {
					ROOT = THIS
				}
			}
			change_relations = {
				country = ROOT
				value = 2
			}
		}
	}
}

	trigger = {
	}

	immediate = {
		var:curie_physicist_var = {
			save_scope_as = curie_physicist_scope
		}
		random_state = {
			limit = {
				state_region = s:STATE_HANNOVER
			}
			owner = {
				save_scope_as = curie_goettingen_owner
			}
		}
		if = {
			limit = { has_variable = curie_physicist_var }
			var:curie_physicist_var = {
				set_variable = {
					name = curie_physicist_achievements_finished_geomagnetic_observatory_project_var
					value = flag:curie_physicist_achievements_finished_geomagnetic_observatory_project_loc
				}
			}
		}
	}

	option = {
		name = curie.102.g #We should build a measuring station immediately.
		default_option = yes
		mathematical_physicist_experience_gain_effect = { AMOUNT = 30 }
		scope:curie_physicist_scope = {
			set_character_busy_and_immortal = no
		}
		curie_add_x_magnetic_data = {
			AMOUNT = 5
		}
		curie_add_x_physical_data_effect = {
			AMOUNT = 2
		}
		random_scope_state = {
			limit = {
				has_building = building_verrier_observatory
			}
			activate_production_method = {
				building_type = building_verrier_observatory
				production_method = pm_curie_magnetic_observatory
			}
		}
	}
}

	type = country_event

	title = curie.104.t
	desc = curie.104.d
	flavor = curie.104.f

	gui_window = event_window_2char

	left_icon = scope:curie_physicist_scope
	right_icon = scope:gaudi_admiral_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	cooldown = {
		years = 5
	}

	trigger = {
		any_scope_character = {
			has_role_of_type = admiral
			is_busy = no
		}
	}

	immediate = {
		random_scope_character = {
			limit = {
				has_role_of_type = admiral
				is_busy = no
			}
			save_scope_as = curie_admiral_scope
		}
		var:curie_physicist_var = {
			save_scope_as = curie_physicist_scope
		}
	}

	option = {
		name = curie.104.g #The new knowledge should be useful to him.
		default_option = yes
		scope:curie_admiral_scope = {
			add_modifier = {
				name = curie_geomagnetical_knowledge_modifier
			}
		}
	}

	option = {
		name = curie.104.h #This knowledge should benefit everyone.
		curie_add_x_physical_data_effect = {
			AMOUNT = 2
		}
		curie_add_x_magnetic_data = {
			AMOUNT = 1
		}
	}
}
