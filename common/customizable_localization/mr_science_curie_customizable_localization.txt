curie_magnetic_association_city = {
	type = country

	text = {
		trigger = {
			global_var:curie_magnetic_association_city_global_var ?= s:STATE_HANNOVER
		}
		localization_key = curie_magnetic_association_city_gottingen
	}

	text = {
		trigger = {

	}

	text = {
		localization_key = curie_magnetic_association_city_generic
	}
