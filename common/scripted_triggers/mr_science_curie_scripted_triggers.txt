######Physicist CHARACTER TRIGGERS######

curie_has_physicist = {
	custom_tooltip = {
		text = curie_has_physicist_tt
		curie_has_physicist_no_tooltip_trigger = yes
	}
}
curie_has_physicist_no_tooltip_trigger = {
	has_variable = curie_physicist_var
}

curie_has_active_physicist_trigger = {
	custom_tooltip = {
		text = curie_has_active_physicist_trigger_tt
		curie_has_active_physicist_no_tooltip_trigger = yes
	}
}
curie_has_active_physicist_no_tooltip_trigger = {
	has_variable = curie_physicist_var
	var:curie_physicist_var ?= {
		is_busy = no
	}
}

#Uranium state

curie_is_uranium_state = {
    any_scope_building = {
		is_building_type = building_uranium_mine
	}
}

curie_is_physics_automated_trigger = {
	OR = {
		is_ai = yes
		has_variable = curie_physics_automation_var
	}
}

#Trigger to check if a specific formula was not discovered
curie_formula_x_is_not_discovered_trigger = {
	#NUMBER --> Number of the Element in the Formula List
#	hidden_trigger = {
		save_temporary_scope_value_as = { #This is used to display the amount in the tooltip
			name = value
			value = $NUMBER$
		}
#	}

	custom_tooltip = {
		text = curie_formula_x_is_not_discovered_tt
		NOT = {
			curie_formula_x_is_discovered_no_tooltip_trigger = { NUMBER = $NUMBER$ }
		}
	}
}

#Check if the formula in question is discovered or not

curie_formula_x_is_not_discovered_no_tooltip_trigger = {
	#NUMBER --> Number of the Element in the Periodic Table
	NOT = {
		curie_formula_x_is_discovered_no_tooltip_trigger = { NUMBER = $NUMBER$ }
	}
}

curie_formula_x_is_discovered_no_tooltip_trigger = {
	any_in_global_list = {
		variable = curie_formulas_global_list

		var:curie_formula_number_var ?= $NUMBER$
		has_variable = curie_formula_discovered_var
	}
}
