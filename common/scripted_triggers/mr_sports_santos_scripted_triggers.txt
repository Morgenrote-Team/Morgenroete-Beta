#Has Balloonist
santos_has_balloonist_trigger = {
	custom_tooltip = {
		text = has_balloonist_trigger_tt
		santos_has_balloonist_no_tooltip_trigger = yes
	}
}
santos_has_balloonist_no_tooltip_trigger = {
	has_variable = santos_balloonist_var
}
santos_no_balloonist_trigger = {
	custom_tooltip = {
		text = has_no_balloonist_trigger_tt
		santosno_balloonist_no_tooltip_trigger = yes
	}
}
santos_no_balloonist_no_tooltip_trigger = {
	NOT = {
		has_variable = santos_balloonist_var
	}
}

santos_has_active_balloonist_trigger = {
	custom_tooltip = {
		text = santos_has_active_balloonist_trigger_tt
		santos_has_active_balloonist_no_tooltip_trigger = yes
	}
}
santos_has_active_balloonist_no_tooltip_trigger = {
	has_variable = santos_balloonist_var
	var:santos_balloonist_var ?= {
		is_busy = no
	}
}

santos_balloon_crossing_goal_trigger = {
	OR = {
		custom_tooltip = {
			text = je_santos_balloon_crossing_small_complete_tt
			AND = {
				OR = {
					var:santos_balloon_crossing_target_var ?= flag:santos_channel_flag
					var:santos_balloon_crossing_target_var ?= flag:santos_rio_flag
					var:santos_balloon_crossing_target_var ?= flag:santos_korea_flag
				}
				var:santos_balloon_crossing_progress_var >= 8
			}
		}
		custom_tooltip = {
			text = je_santos_balloon_crossing_medium_complete_tt
			AND = {
				OR = {
					var:santos_balloon_crossing_target_var ?= flag:santos_skagerrak_flag
					var:santos_balloon_crossing_target_var ?= flag:santos_pyrenees_flag
					var:santos_balloon_crossing_target_var ?= flag:santos_bab_flag
				}
				var:santos_balloon_crossing_progress_var >= 12
			}
		}
		custom_tooltip = {
			text = je_santos_balloon_crossing_large_complete_tt
			AND = {
				OR = {
					var:santos_balloon_crossing_target_var ?= flag:santos_alps_flag
					var:santos_balloon_crossing_target_var ?= flag:santos_mediterranean_flag
					var:santos_balloon_crossing_target_var ?= flag:santos_andes_flag
				}
				var:santos_balloon_crossing_progress_var >= 16
			}
		}
	}
}
