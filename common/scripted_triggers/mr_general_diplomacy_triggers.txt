#Diplomacy

mr_can_have_research_charter_trigger = {
	OR = {
		lepsius_has_excavation_site_for_licence_trigger = yes #Archaeological Expedition
		ito_country_is_potential_botanical_expedition_target_trigger = yes #Botanical Expedition
		dubois_has_great_formation = yes #Paleontological Expedition
		douglas_has_conquerable_summit_trigger = yes #Mountaineering Expedition
	}
}

mr_has_research_charter_trigger = {
	any_scope_treaty = {
		binds = ROOT
		any_scope_article = {
			has_type = mr_research_charter
			target_country = ROOT
		}

		source_country = {
			NOT = {
				has_war_with = ROOT
			}
		}
	}
	#In case that we have to revive the old system, we should use this:
	#has_diplomatic_pact = {
	#	who = ROOT
	#	type = mr_research_license
	#}
}

mr_research_charter_ai_matching_opportunity_trigger = {
    OR = {
        # Archaeologist -> Excavation site
        AND = {
            has_variable = lepsius_archaeologist_var
            scope:other_country = {
                lepsius_has_excavation_site_for_licence_trigger = yes
            }
        }

        # Naturalist / Botanist -> Medicinal plants
        AND = {
            has_variable = ito_naturalist_var
            scope:other_country = {
                ito_country_is_potential_botanical_expedition_target_trigger = yes
            }
        }

        # Paleontologist -> Geological formation
        AND = {
            has_variable = dubois_paleontologist_var
            scope:other_country = {
                dubois_has_great_formation = yes
            }
        }

        # Mountaineer -> Conquerable summit
        AND = {
            has_variable = douglas_mountaineer_var
            scope:other_country = {
                douglas_has_conquerable_summit_trigger = yes
            }
        }
    }
}
