#Character with negative Trait
mr_character_has_negative_trait_trigger = {
	OR = {
		has_trait = alcoholic
		has_trait = expensive_tastes
		has_trait = opium_addiction
		has_trait = cocaine_addiction
		has_trait = syphilis
		has_trait = psychological_affliction
		has_trait = panum_cholera
		has_trait = tuberculosis
		has_trait = panum_yellow_fever
	}
}

#Character with Disease Trait
mr_character_has_disease_trait_trigger = {
	OR = {
		has_trait = syphilis
		has_trait = psychological_affliction
		has_trait = senile
		has_trait = panum_cholera
		has_trait = tuberculosis
		has_trait = panum_yellow_fever
	}
}

#Character with Addiction Trait
mr_character_has_addiction_trait_trigger = {
	OR = {
		has_trait = alcoholic
		has_trait = cocaine_addiction
		has_trait = opium_addiction
	}
}

#Character with Addiction or Disease Trait
mr_character_has_addiction_or_disease_trait_trigger = {
	OR = {
		mr_character_has_addiction_trait_trigger = yes
		mr_character_has_disease_trait_trigger = yes
	}
}

#Checks if a character is alive and not busy
mr_is_character_alive_and_not_busy_trigger = {
	is_character_alive = yes
	is_busy = no
}
