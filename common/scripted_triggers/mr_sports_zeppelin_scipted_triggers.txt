######ZEPPELIN CAPTAIN######

zeppelin_has_captain = {
	custom_tooltip = {
		text = zeppelin_has_captain_tt
		zeppelin_has_captain_no_tooltip_trigger = yes
	}
}
zeppelin_has_captain_no_tooltip_trigger = {
	has_variable = zeppelin_captain_var
}

curtiss_has_active_captain = {
	custom_tooltip = {
		text = zeppelin_has_active_captain_tt
		curtiss_has_active_captain_no_tooltip_trigger = yes
	}
}
curtiss_has_active_captain_no_tooltip_trigger = {
	has_variable = zeppelin_captain_var
	var:zeppelin_captain_var ?= {
		mr_is_character_alive_and_not_busy_trigger = yes
	}
}

######ZEPPELIN######

zeppelin_is_active_trigger = {
	custom_tooltip = {
		text = zeppelin_is_active_trigger_tt
		var:zeppelin_captain_var ?= { var:zeppelin_status_var ?= flag:zeppelin_status_active_flag }
	}
}

zeppelin_is_inactive_trigger = {
	custom_tooltip = {
		text = zeppelin_is_inactive_trigger_tt
		var:zeppelin_captain_var ?= { var:zeppelin_status_var ?= flag:zeppelin_status_inactive_flag }
	}
}

zeppelin_has_hangar_trigger = {
	custom_tooltip = {
		text = zeppelin_has_hangar_trigger_tt
		any_scope_building = {
			is_building_type = building_zeppelin_hangar
		}
	}
}

zeppelin_no_active_zeppelin_construction_trigger = {
	custom_tooltip = {
		text = zeppelin_no_active_zeppelin_construction_trigger_tt
		NOT = {
			has_journal_entry = je_zeppelin_construction
		}
	}
}

zeppelin_construction_medium_possible_trigger = {
	mr_gui_has_art_silk_tech_trigger = yes
}

zeppelin_construction_large_possible_trigger = {
	mr_gui_has_art_silk_tech_trigger = yes
	zeppelin_has_hangar_trigger = yes
}

zeppelin_construction_giant_possible_trigger = {
	mr_gui_has_art_silk_tech_trigger = yes
	zeppelin_has_hangar_trigger = yes
	tesla_has_aeronautical_engineer_trigger = yes
}

zeppelin_size_medium_trigger = {
	custom_tooltip = {
		text = zeppelin_size_medium_trigger_tt
		OR = {
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_medium_flag }
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_large_flag }
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_giant_flag }
		}
	}
}

zeppelin_size_large_trigger = {
	custom_tooltip = {
		text = zeppelin_size_large_trigger_tt
		OR = {
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_large_flag }
			var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_giant_flag }
		}
	}
}

zeppelin_size_giant_trigger = {
	custom_tooltip = {
		text = zeppelin_size_giant_trigger_tt
		var:zeppelin_captain_var ?= { var:zeppelin_size_var ?= flag:zeppelin_size_giant_flag }
	}
}

zeppelin_construction_goal_trigger = {
	OR = {
		AND = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_small_flag
			je:je_zeppelin_construction = {
				"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 6
			}
		}
		AND = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_medium_flag
			je:je_zeppelin_construction = {
				"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 12
			}
		}
		AND = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_large_flag
			je:je_zeppelin_construction = {
				"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 30
			}
		}
		AND = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_giant_flag
			je:je_zeppelin_construction = {
				"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 60
			}
		}
	}
}

zeppelin_captain_creation_trigger = {
	NOR = {
		has_variable = zeppelin_new_captain_var
		has_variable = zeppelin_captain_var
	}
	AND = {
		var:zeppelin_construction_target_var ?= flag:zeppelin_small_flag
		je:je_zeppelin_construction = {
			"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 4
		}
	}
	AND = {
		var:zeppelin_construction_target_var ?= flag:zeppelin_medium_flag
		je:je_zeppelin_construction = {
			"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 8
		}
	}
	AND = {
		var:zeppelin_construction_target_var ?= flag:zeppelin_large_flag
		je:je_zeppelin_construction = {
			"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 24
		}
	}
	AND = {
		var:zeppelin_construction_target_var ?= flag:zeppelin_giant_flag
		je:je_zeppelin_construction = {
			"scripted_bar_progress(zeppelin_construction_progress_bar)" >= 52
		}
	}
}
