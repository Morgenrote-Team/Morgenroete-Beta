zeppelin_has_hangar_trigger = {
	custom_tooltip = {
		text = zeppelin_has_hangar_trigger_tt
		any_scope_building = {
			is_building_type = building_zeppelin_hangar
		}
	}
}

zeppelin_no_active_zeppelin_construction_trigger = {
	custom_tooltip = {
		text = zeppelin_no_active_zeppelin_construction_trigger_tt
		NOT = {
			has_journal_entry = je_zeppelin_construction
		}
	}
}

zeppelin_construction_medium_possible_trigger = {
	mr_gui_has_art_silk_tech_trigger = yes
}

zeppelin_construction_large_possible_trigger = {
	mr_gui_has_art_silk_tech_trigger = yes
	zeppelin_has_hangar_trigger = yes
}

zeppelin_construction_giant_possible_trigger = {
	mr_gui_has_art_silk_tech_trigger = yes
	zeppelin_has_hangar_trigger = yes
	tesla_has_aeronautical_engineer_trigger = yes
}
