######STOP CURRENT PHYSICIST PROJECT######

curie_stop_physicist_project_button = {
	name = "curie_stop_physicist_project_button"
	desc = "curie_stop_physicist_project_button_desc"

	visible = {  always = yes }

	possible = {
		NOT = { has_variable = curie_stop_physicist_improvement_project_var }
	}

	effect = {
		set_variable = {
			name = curie_stop_physicist_improvement_project_var
			days = 7
		}
	}

	ai_chance = {
		value = 0 #It's just not worth for you, Mr AI
	}
}

###### MATHEMATICS IMPROVEMENT PROJECT ######

je_curie_improve_mathematics_project_change_list_button = {
	name = "je_curie_improve_mathematics_project_change_list_button"
	desc = "je_curie_improve_mathematics_project_change_list_button_desc"

	possible = {
		custom_tooltip = {
			text = MR_ALREADY_ADDING_NEW_BUILDINGS_TO_LIST_TRIGGER_TT
			NOT = {
				has_variable = curie_improve_mathematics_project_building_choser_var
			}
		}
	}

	effect = {
		clear_variable_list = curie_improve_mathematics_buildings_variable_list
		ordered_scope_building = {
			limit = {
				curie_building_valid_for_mathematics_project_no_tooltip_trigger = yes
				NOT = {
					root = {
						is_target_in_variable_list = {
							name = curie_improve_mathematics_buildings_chosen_variable_list
							target = prev
						}
					}
				}
			}
			order_by = level
			check_range_bounds = no

			root = {
				add_to_variable_list = {
					name = curie_improve_mathematics_buildings_variable_list
					target = prev
				}
			}
		}
		#Activates the window
		set_variable = curie_improve_mathematics_project_building_choser_var
	}

	ai_chance = {
		value = 0 #It's just not worth for you, Mr AI
	}
}
