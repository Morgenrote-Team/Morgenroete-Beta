#Access to the Zeppelin Hangar
mr_zeppelin_hangar_button_sgui = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		zeppelin_has_captain = yes
		zeppelin_no_active_zeppelin_construction_trigger = yes
        hidden_trigger = {
            NOT = {
				has_variable = zeppelin_construction_selection_var
			}
        }
	}

	effect = {

	}
}

#Change to Hydrogen
mr_zeppelin_hydrogen_button = {
	scope = country

	is_shown = {
		var:zeppelin_captain_var ?= {
			var:zeppelin_fuel_var ?= flag:zeppelin_fuel_helium_flag
		}
	}

	is_valid = {
		custom_tooltip = {
			text = MR_ZEPPELIN_MODIFICATION_POSSIBLE_TRIGGER_TT
			NOT = {
				has_modifier = zeppelin_modification_modifier
			}
		}
		custom_tooltip = {
			text = MR_BALLOONIST_HYDROGEN_FACTORY_TRIGGER_TT
			curtiss_has_hydrogen = yes
		}
	}

	effect = {
		zeppelin_set_zeppelin_fuel_hydrogen_effect = yes
	}
}

			text = MR_BALLOONIST_HELIUM_FACTORY_TRIGGER_TT

		base = 100 # The chance that the AI will activate this SGUI (script value between 1 and 100)
	}
	ai_frequency = 6 # Named value determining how frequently the AI will evaluate this SGUI (in months)
}
