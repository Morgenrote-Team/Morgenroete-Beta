#Visibility of the Mathematician Experience Bar
curie_mathematical_physicist_experience_bar_visibility_sgui = {
	scope = country
	is_shown = {
		OR = {
			AND = {
				has_variable = curie_physicist_var
				var:curie_physicist_var ?= {
					has_modifier = academics_is_mathematical_physicist_modifier
				}
			}
			NOT = { has_variable = curie_physicist_var }
		}
	}
}

### Change to other Physicist Specialization ###

#MR_CHANGE_TO_MATHEMATICAL_PHYSICIST_BUTTON_TT
curie_change_to_mathematical_physicist_button_sgui = {
	scope = country
	is_valid = {
		custom_tooltip = {
			text = MR_HAS_PHYSICIST_TT
			curie_has_physicist = yes
		}
		custom_tooltip = {
			text = MR_CHANGE_TO_MATHEMATICAL_PHYSICIST_BUTTON_HAS_NO_MATHEMATICAL_PHYSICIST_TT
			curie_has_no_mathematical_physicist_trigger = yes
		}

		custom_tooltip = {
			text = MR_CHANGE_TO_MATHEMATICAL_PHYSICIST_BUTTON_NO_RECENT_SPECIALIZATION_SWITCH_TT
			var:curie_physicist_var ?= {
				NOT = { has_modifier = academics_physicist_recent_specialization_switch_modifier }
			}
		}
	}
	effect = {
		curie_set_mathematical_physicist_effect = yes
		if = {
			limit = {
				has_variable = curie_physicist_var
			}
			var:curie_physicist_var = {
				add_modifier = {
					name = academics_physicist_recent_specialization_switch_modifier
					years = 5
				}
			}
		}
	}
}

#MR_STOP_PHYSICS_PROJECT_BUTTON
mr_stop_physics_project_button_sgui = {
	scope = country
	is_valid = {
		custom_tooltip = {
			text = MR_STOP_PHYSICS_PROJECT_POSSIBLE
			has_variable = curie_can_stop_active_physics_project_var

		NOT = { has_variable = curie_stop_physics_project_var }
	}
	effect = {
		set_variable = curie_stop_physics_project_var
	}
}

### Physicist Automation ###

mr_physicist_automation_gui = {
	scope = country
	is_shown = {
		NOT = { has_variable = curie_physics_automation_var }
	}
	is_valid = {
		always = yes
	}
	effect = {
		custom_tooltip = mr_physicist_automation_gui_tt
		set_variable = curie_physics_automation_var
		ai_physics_automation_effect = yes
	}
}
mr_stop_physicist_automation_gui = {
	scope = country
	is_shown = {
		has_variable = curie_physics_automation_var
	}
	is_valid = {
		always = yes
	}
	effect = {
		custom_tooltip = mr_stop_physicist_automation_gui_tt
		remove_variable = curie_physics_automation_var
	}
}
