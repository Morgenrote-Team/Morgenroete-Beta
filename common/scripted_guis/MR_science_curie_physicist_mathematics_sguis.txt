###Magnetic Association###

#MR_PHYSICIST_FOUND_MAGNETIC_ASSOCIATION_BUTTON
mr_physicist_found_magnetic_association_button = {
	scope = country
	is_valid = {
		curie_has_physicist_trigger = yes
		mr_gui_has_differential_geometry_tech_trigger = yes
		custom_tooltip = {
			text = MR_ASTEROID_CONGRESS_OBSERVATORY_BUILDING_TRIGGER_TT
			any_scope_building = {
				is_building_type = building_verrier_observatory
				level > 0
			}
		}
		custom_tooltip = {
			text = MR_NO_MAGNETIC_ASSOCIATION_TRIGGER_TT
			NOT = {
				has_global_variable = curie_magnetic_association_global_var
			}
		}
	}
	effect = {
		curie_set_magnetic_association_effect = yes
		trigger_event = {
			id = curie.100
			popup = yes
		}
	}

	ai_is_valid = { always = yes }	# Whether this SGUI is available to the AI or not (default false)
	ai_chance = {
		base = 100 # The chance that the AI will activate this SGUI (script value between 1 and 100)
	}
	ai_frequency = 6 # Named value determining how frequently the AI will evaluate this SGUI (in months)
}

#MR_PHYSICIST_MAGNETIC_RESEARCH_STATION_BUTTON
mr_physicist_magnetic_research_station_button = {
	scope = country
	is_valid = {
		curie_has_active_physicist_trigger = yes
		custom_tooltip = {
			text = MAGNETIC_ASSOCIATION_TRIGGER_TT
			has_global_variable = curie_magnetic_association_global_var
		}
		custom_tooltip = {
			text = MR_PHYSICIST_NO_MAGNETIC_OBSERVATORY_TRIGGER_TT
			any_scope_building = {
				is_building_type = building_verrier_observatory
				has_active_production_method = pm_curie_no_magnetic_observatory
				level > 0
			}
		}
		mr_gui_has_differential_geometry_tech_trigger = yes
		NOT = {
			has_variable = curie_magnetic_station_var
		}
	}
	effect = {
		if = {
			limit = {
				NOT = {
					has_variable = curie_magnetic_association_member_var
				}
			}
			trigger_event = {
				id = curie.101
				popup = yes
			}
		}
		var:curie_physicist_var ?= {
			set_character_busy_and_immortal = yes
		}
		curie_add_magnetic_association_member = yes
		add_journal_entry = {
			type = je_curie_magnetic_station
		}
	}

	ai_is_valid = { always = yes }	# Whether this SGUI is available to the AI or not (default false)
	ai_chance = {
		base = 100 # The chance that the AI will activate this SGUI (script value between 1 and 100)
	}
	ai_frequency = 6 # Named value determining how frequently the AI will evaluate this SGUI (in months)
}

#Active
mr_physicist_magnetic_research_station_active = {
	scope = country
	is_shown = {
		has_journal_entry = je_curie_magnetic_station
	}
}

#MR_PHYSICIST_MAGNETIC_ATLAS_BUTTON
mr_physicist_magnetic_atlas_button = {
	scope = country
	is_valid = {
		curie_has_active_physicist_trigger = yes
		custom_tooltip = {
			text = MR_PHYSICIST_MAGNETIC_OBSERVATORY_TRIGGER_TT
			any_scope_building = {
				is_building_type = building_verrier_observatory
				has_active_production_method = pm_curie_magnetic_observatory
				level > 0
			}
		}
		custom_tooltip = {
			text = MR_PHYSICIST_15_MAGNETIC_DATA_TRIGGER_TT
			var:curie_magnetic_data_var >= 15
		}
		NOT = {
			has_variable = curie_magnetic_atlas_var
		}
	}
	effect = {
		var:curie_physicist_var ?= {
			set_character_busy_and_immortal = yes
		}
		add_journal_entry = {
			type = je_curie_magnetic_atlas
		}
	}

	ai_is_valid = { always = yes }	# Whether this SGUI is available to the AI or not (default false)
	ai_chance = {
		base = 100 # The chance that the AI will activate this SGUI (script value between 1 and 100)
	}
	ai_frequency = 6 # Named value determining how frequently the AI will evaluate this SGUI (in months)
}

#Active
mr_physicist_magnetic_atlas_active = {
	scope = country
	is_shown = {
		has_journal_entry = je_curie_magnetic_atlas
	}
}
