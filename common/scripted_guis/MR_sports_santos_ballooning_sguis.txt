#Does the country have a balloonist
mr_has_balloonist_sgui = {
	scope = country
	is_shown = {
		has_variable = santos_balloonist_var
	}
}

######SPONSOR BALLOONIST######
mr_sponsor_balloonist_button_sgui = {
    scope = country
	
	is_shown = {
		santos_no_balloonist_no_tooltip_trigger = yes
	}

    is_valid = {
		mr_gui_has_montgolfier_balloon_tech_trigger = yes
    }

    effect = {
		custom_tooltip = MR_BALLOONIST_COST_TT
		trigger_event = {
			id = santos.650
			popup = yes
		}
    }
}

######FIRE BALLOONIST######
mr_fire_balloonist_button_sgui = {
    scope = country
	
	is_shown = {
		santos_has_balloonist_trigger = yes
	}

    is_valid = {
		always = yes
    }

    effect = {
		set_variable = {
			name = santos_balloonist_fired_var
			days = 5
		}
		retire_balloonist_effect = yes
    }
}

#MR_BALLOONIST_BALLOON_RIDE_BUTTON
mr_balloonist_balloon_ride_button = {
	scope = country

	is_shown = {
		has_variable = santos_balloonist_var
	}

	is_valid = {
		santos_has_balloonist_trigger = yes
		NOT = {
			has_journal_entry = je_santos_balloon_ride
		}
	}
	effect = {
		add_journal_entry = { type = je_santos_balloon_ride }
	}
}
#Active
mr_balloonist_balloon_ride_active = {
	scope = country
	is_shown = {
		has_journal_entry = je_santos_balloon_ride
		santos_has_active_balloonist_no_tooltip_trigger = yes
	}
}

#MR_BALLOONIST_BALLOON_ADVENTURE_BUTTON
mr_balloonist_balloon_adventure_button = {
	scope = country

	is_shown = {
		has_variable = santos_balloonist_var
	}

	is_valid = {
		santos_has_balloonist_trigger = yes
		mr_gui_has_research_charter_tech_trigger = yes
		NOT = {
			has_journal_entry = je_santos_balloon_adventure
		}
	}
	effect = {
		var:santos_balloonist_var ?= {
			set_character_busy_and_immortal = no
		}
		add_journal_entry = { type = je_santos_balloon_adventure }
	}
}
#Active
mr_balloonist_balloon_adventure_active = {
	scope = country
	is_shown = {
		has_journal_entry = je_santos_balloon_adventure
	}
}

######Gordon Bennett Cup######
mr_gordon_bennett_cup_button_sgui = {
    scope = country
	
	is_shown = {
		NOT = {
			has_global_variable = santos_gordon_bennett_cup_found_global_var
		}
		santos_has_balloonist_trigger = yes
	}

    is_valid = {
		mr_gui_has_vikelas_international_sports_tech_trigger = yes
		mr_gui_has_early_aviation_tech_trigger = yes
		custom_tooltip = {
			text = MR_GORDON_BENNETT_PUBLISHER_BUILDING_TRIGGER_TT
			country_has_building_type_levels = {  
				target = bt:building_manzoni_publishing_industry
				value  >= 10
			}
		}
    }

    effect = {
		set_global_variable = santos_gordon_bennett_cup_found_global_var
		trigger_event = {
			id = santos.681
			popup = yes
		}
    }
}

### Change Fuel ###

#Change to Coal Gas
mr_balloonist_coal_gas_button = {
    scope = country
	
	is_shown = {
		var:santos_balloonist_var ?= {
			var:santos_balloon_fuel_var ?= flag:santos_balloon_fuel_hydrogen_flag
		}
	}

    is_valid = {
		custom_tooltip = {
			text = MR_BALLOONIST_MODIFICATION_POSSIBLE_TRIGGER_TT
			NOT = {
				has_modifier = santos_balloon_modification_modifier
			}
		}
    }

    effect = {
		santos_set_balloon_fuel_coal_gas_effect = yes
    }
}

#Change to Hydrogen
mr_balloonist_hydrogen_button = {
    scope = country
	
	is_shown = {
		var:santos_balloonist_var ?= {
			var:santos_balloon_fuel_var ?= flag:santos_balloon_fuel_coal_gas_flag
		}
	}

    is_valid = {
		custom_tooltip = {
			text = MR_BALLOONIST_MODIFICATION_POSSIBLE_TRIGGER_TT
			NOT = {
				has_modifier = santos_balloon_modification_modifier
			}
		}
		custom_tooltip = {
			text = MR_BALLOONIST_HYDROGEN_FACTORY_TRIGGER_TT
			any_scope_state = {
				any_scope_building = {
					is_building_type = building_explosives_factory
					has_active_production_method = pm_brine_electrolysis
				}
			}
		}
    }

    effect = {
		santos_set_balloon_fuel_hydrogen_effect = yes
    }
}

### Change Type ###

#Change Type
mr_balloonist_change_type_button = {
    scope = country
	
	is_shown = {
		always = yes
	}

    is_valid = {
		custom_tooltip = {
			text = MR_BALLOONIST_MODIFICATION_POSSIBLE_TRIGGER_TT
			NOT = {
				has_modifier = santos_balloon_modification_modifier
			}
		}
		hidden_trigger = {
			NOT = {
				has_variable = santos_balloon_type_event_active_var
			}
		}
    }

    effect = {
		trigger_event = {
			id = santos.691
			popup = yes
		}
    }
}
