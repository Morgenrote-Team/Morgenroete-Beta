mr_aus_mendel_gregor_character_template = {
	dna = mr_aus_mendel_gregor
	first_name = mr_gregor
	last_name = mr_mendel
	culture = cu:south_german
	role = character_role_theiler_biologist
	role = character_role_curie_physicist
	interest_group = ig_devout
	birth_date = 1822.07.20
	historical = yes
	home_region = STATE_MORAVIA #Born in Heinzendorf bei Odrau (today's Hynčice)
	traits = {
		theiler_biologist
		ito_botanist
		curie_physicist
		curie_meteorologist
		verrier_mathematician
		verrier_astronomer
		meticulous
		innovative
		erudite
	}
	on_created = {
		set_variable = mr_no_headgear_var
		set_variable = mr_female_glasses_var
		set_variable = mr_devout_clothing
		mr_make_character_biologist_effect = yes
		mr_make_character_physicist_effect = yes
		set_global_variable = mr_mendel_gregor_created_global_var
	}
}
