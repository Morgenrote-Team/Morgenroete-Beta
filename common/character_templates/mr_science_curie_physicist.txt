	role = character_role_curie_physicist
	interest_group = ig_intelligentsia
	birth_date = 1803.11.29
	historical = yes
	home_region = STATE_TYROL #Born in Salzburg
	traits = {
		curie_physicist
		curie_mathematical_physicist
		verrier_astronomer
		innovative
		meticulous
		erudite

		set_global_variable = mr_doppler_christian_created_global_var
