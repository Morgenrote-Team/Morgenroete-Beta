curtiss_seaplanes_tech = {

	era = era_5
	texture = "gfx/interface/icons/invention_icons/curtiss_seaplanes_tech.dds"
	category = military

		military_aviation
		dreadnought_tech
	}

	ai_weight = {
		value = 1

			limit = { navy_size >= 20 }
			add = 0.5

			}

		}

		}
