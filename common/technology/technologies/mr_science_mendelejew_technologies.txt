mendelejew_natural_gas_tech = {
	era = era_4
	texture = "gfx/interface/icons/invention_icons/mendelejew_natural_gas_tech.dds"
	category = production

	unlocking_technologies = {
		pumpjacks
	}

	modifier = {
		goods_output_oil_mult = 0.15
	}

	ai_weight = {
		value = 1
		if = {
			limit = {
				OR = {
					has_strategy = ai_strategy_industrial_expansion
					has_strategy = ai_strategy_resource_expansion
				}
			}
			add = 1
		}
		if = {
			limit = {
				any_scope_building = {
					is_building_type = building_oil_rig
				}
			}
			add = 2
		}
		if = {
			limit = {
				any_scope_state = {
					has_variable = agassiz_has_helium_var
				}
			}
			add = 2
		}
	}
}
