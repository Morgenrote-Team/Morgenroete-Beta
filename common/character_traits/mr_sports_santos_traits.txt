santos_balloonist = {
	type = skill
	texture = "gfx/interface/icons/character_trait_icons/santos_balloonist.dds"

	character_modifier = {
		character_popularity_add = 5
	}

	possible = {
		owner ?= {
			has_technology_researched = curtiss_montgolfier_balloon_tech
		}
		NOT = { age < define:NCharacters|ADULT_AGE }
	}

	weight = {
		value = 0.5
		if = {
			limit = {
				has_trait = brave
			}
			multiply = 4
		}
	}

	replace = {
	}

	value = 1
}
