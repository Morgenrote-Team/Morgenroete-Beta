# This effect is used to set up whether the Morgenröte Characters should wear academic gowns or normal clothes
mr_set_randomized_academic_clothing_effect = {
	random_list = {
		50 = {
			set_variable = mr_normal_clothes_var

			modifier = {
				add = {
					value = year
					subtract = global_var:start_year_global_var
					divide = 2
					max = 50
					min = 0
				}
			}
		}
		50 = {
			set_variable = mr_academic_clothes_var

			modifier = {
				subtract = {
					value = year
					subtract = global_var:start_year_global_var
					divide = 2
					max = 50
					min = 0
				}
			}
		}
	}
}
