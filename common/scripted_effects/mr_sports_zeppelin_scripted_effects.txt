zeppelin_set_zeppelin_fuel_hydrogen_effect = {
	custom_tooltip = {
	text = zeppelin_set_zeppelin_fuel_hydrogen_effect_tt
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_fuel_var
				value = flag:zeppelin_fuel_hydrogen_flag
			}
		}
		add_modifier = {
			name = zeppelin_modification_modifier
			days = 180
		}
		set_variable = {
			name = zeppelin_fuel_var
			value = 0
		}
	}
}

#Effect that gets fired when a country has reached the progress (8/12/16)
zeppelin_construction_finished_effect = {
	zeppelin_set_name_effect = yes
	zeppelin_set_size_effect = yes
	zeppelin_set_active_effect = yes
	zeppelin_set_maiden_voyage_effect = yes
	trigger_event = {
		id = zeppelin.131
	}
}

zeppelin_set_size_effect = {
	#Small Zeppelin
	if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_small_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_small_flag
			}	
		}
	}
	#Medium-sized Zeppelin
	else_if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_medium_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_medium_flag
			}	
		}
	}
	#Large Zeppelin
	else_if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_large_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_large_flag
			}	
		}
	}
	#Giant Zeppelin
	else_if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_giant_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_giant_flag
			}
		}
	}
}

zeppelin_set_active_effect = {

		}
	}
}

zeppelin_set_name_effect = {
	set_variable = {
		name = zeppelin_name_randomizer_var
		value = { 1 20 }
	}

		name = zeppelin_name_randomizer_var
		nearest = 1
	}
}
