zeppelin_map_arctic_region_effect = {

	#save_temporary_scope_value_as = { #This is used to display the amount in the tooltip
	#	name = value
	#	value = $REGION$
	#}

	if = {
		limit = {
			NOT = {
				has_variable = zeppelin_arctic_region_num_var
			}
		}
		set_variable = {
			name = zeppelin_arctic_region_num_var
			value = 0
		}
	}
	
	change_variable = {
		name = zeppelin_arctic_region_num_var
		add = 1
	}

}

### Event Log ###

#Clears all Event log containers
zeppelin_expedition_clear_event_log_containers_je_scope_effect = {
	every_container = {
		limit = {
			has_tag = zeppelin_expedition_event
		}
		destroy_container = yes
	}
	clear_variable_list = zeppelin_expedition_events_list
}

#Creates a new container which is used in the JE to display recent events.
# Execute in the Journal Entry scope.
# An event in this case is a short text which will be displayed in a list.
zeppelin_expedition_event_list_add_container_entry_effect = {
	# ICON = Icon reference, in our case a flag which references a loc key
	# TITLE_LOCALIZATION_KEY = Title localization
	# TOOLTIP_KEY = Localization key for the event description
	# PRIMARY_COUNTRY = Country to reference in event description
	# ADDITIONAL_EFFECTS_CONTAINER = Additional effects in the container scope
	create_container = {
		tags = { mr_mod zeppelin_expedition_event }
		parent = this
		save_scope_as = zeppelin_event_container_scope
	}

	scope:zeppelin_event_container_scope ?= {
		set_variable = {
			name = event_container_icon
			value = $ICON$
		}
		set_variable = {
			name = event_container_primary_country
			value = $PRIMARY_COUNTRY$
		}
		set_variable = {
			name = event_container_title_loc
			value = $TITLE_LOCALIZATION_KEY$
		}
		set_variable = {
			name = event_container_tt
			value = $TOOLTIP_KEY$
		}
		set_variable = {
			name = event_container_date
			value = game_date
		}
		$ADDITIONAL_EFFECTS_CONTAINER$
	}
	add_to_variable_list = {
		name = zeppelin_expedition_events_list
		target = scope:zeppelin_event_container_scope
	}
}

zeppelin_country_speed_record_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = zeppelin_speed_record_var
			}
		}
		set_variable = {
			name = zeppelin_speed_record_var
			value = var:new_zeppelin_speed_var
		}
	}
	
	if = {
		limit = {
			var:new_zeppelin_speed_var > var:zeppelin_speed_record_var
		}
		set_variable = {
			name = zeppelin_speed_record_var
			value = var:new_zeppelin_speed_var
		}
		captain_experience_gain_effect = { AMOUNT = 10 } #if country record is broken, a total of 10 experience is added
	}
	if = {
		limit = {
			var:new_zeppelin_speed_var > global_var:zeppelin_speed_record_global_var
		}
		set_global_variable = {
			name = zeppelin_speed_record_global_var
			value = var:new_zeppelin_speed_var
		}
		set_global_variable = {
			name = zeppelin_speed_record_holder_global_var
			value = root
		}
		save_scope_as = zeppelin_record_breaking_country

		if = {
			limit = { NOT = { has_modifier = zeppelin_speed_record_modifier } }
			every_country = { #removes modifier from old country
				limit = {
					has_modifier = zeppelin_speed_record_modifier
				}
				#post_notification = zeppelin_speed_record_broken_notification
				remove_modifier = zeppelin_speed_record_modifier
			}
			add_modifier = { #adds modifier to new country
				name = zeppelin_speed_record_modifier
			}
		}

		captain_experience_gain_effect = { AMOUNT = 25 } #if world record is broken, a total of 25 experience is added
	}
}
