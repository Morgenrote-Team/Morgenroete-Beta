	add_journal_entry = {
		type = je_zeppelin_ballast_invention
	}

		set_character_busy_and_immortal = yes
	}
}

zeppelin_mooring_button_effect = {
	zeppelin_set_inactive_effect = yes
	if = {
		limit = { has_journal_entry = je_zeppelin_military_operation }
		mr_close_journal_entry_effect = { TYPE = je_zeppelin_military_operation }
	}
	if = {
		limit = { has_journal_entry = je_zeppelin_passenger_transport }
		mr_close_journal_entry_effect = { TYPE = je_zeppelin_passenger_transport }
	}
}

#Effect that gets fired when a country has reached the progress (8/12/16)
zeppelin_construction_finished_effect = {
	zeppelin_constructed_by_engineer_effect = yes
	zeppelin_set_name_effect = yes
	zeppelin_set_size_effect = yes
	zeppelin_set_inactive_effect = yes
	zeppelin_set_maiden_voyage_effect = yes
	trigger_event = {
		id = zeppelin.131
	}
}

zeppelin_modification_effect = {
	custom_tooltip = {
		text = zeppelin_modification_effect_tt
		add_modifier = {
			name = zeppelin_modification_modifier
			days = 180
		}
		set_variable = {
			name = zeppelin_fuel_var
			value = 0
		}
	}
}

zeppelin_set_zeppelin_fuel_hydrogen_effect = {
	custom_tooltip = {
		text = zeppelin_set_zeppelin_fuel_hydrogen_effect_tt
		zeppelin_modification_effect = yes
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_fuel_var
				value = flag:zeppelin_fuel_hydrogen_flag
			}
		}
	}
}

zeppelin_constructed_by_engineer_effect = {
	if = {
		limit = {
			tesla_has_aeronautical_engineer_no_tooltip_trigger = yes
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_constructed_by_engineer_var
				value = flag:zeppelin_constructed_by_engineer_flag
			}	
		}
	}
	if = {
		limit = {
			tesla_has_aeronautical_engineer_no_tooltip_trigger = no
			var:zeppelin_captain_var ?= { 
				has_variable = zeppelin_constructed_by_engineer_var
			}
		}
		var:zeppelin_captain_var ?= {
			remove_variable = zeppelin_constructed_by_engineer_var
		}
	}
}

zeppelin_set_name_effect = {
	set_variable = {
		name = zeppelin_name_randomizer_var
		value = { 1 20 }
	}
	round_variable = {
		name = zeppelin_name_randomizer_var
		nearest = 1
	}
}

zeppelin_set_size_effect = {
	#Small Zeppelin
	if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_small_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_small_flag
			}	
		}
	}
	#Medium-sized Zeppelin
	else_if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_medium_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_medium_flag
			}	
		}
	}
	#Large Zeppelin
	else_if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_large_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_large_flag
			}	
		}
	}
	#Giant Zeppelin
	else_if = {
		limit = {
			var:zeppelin_construction_target_var ?= flag:zeppelin_giant_flag
		}
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_size_var
				value = flag:zeppelin_size_giant_flag
			}
		}
	}
}
