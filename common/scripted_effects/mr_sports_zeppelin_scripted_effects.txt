#Effect to add an Arctic Region
zeppelin_map_arctic_region_effect = {

	#This part adds the AMOUNT of National Magnetic Data
	if = { #Fallback if the variable does not exist yet
		limit = {
			NOT = {
				has_variable = zeppelin_arctic_region_var
			}
		}
		set_variable = {
			name = zeppelin_arctic_region_var
			value = 0
		}
	}
	change_variable = {
		name = zeppelin_arctic_region_var
		add = 1
	}

	set_variable = { 
		name = zeppelin_arctic_region_$REGION$_var 
		value = flag:zeppelin_arctic_region_$REGION$_flag 
	}
}

zeppelin_random_arctic_region_effect = {
	if = {
		limit = {
			OR = {
				NOT = {
					has_variable = zeppelin_arctic_region_var
				}
				AND = {
					has_variable = zeppelin_arctic_region_var
					var:zeppelin_arctic_region_var < 9
				}
			}
		}
		custom_tooltip = zeppelin_random_arctic_region_effect_tt
		random_list = {
			1 = { trigger = { zeppelin_arctic_region_nw_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = nw } }
			1 = { trigger = { zeppelin_arctic_region_nn_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = nn } }
			1 = { trigger = { zeppelin_arctic_region_ne_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = ne } }
			1 = { trigger = { zeppelin_arctic_region_ww_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = ww } }
			1 = { trigger = { zeppelin_arctic_region_ee_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = ee } }
			1 = { trigger = { zeppelin_arctic_region_sw_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = sw } }
			1 = { trigger = { zeppelin_arctic_region_ss_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = ss } }
			1 = { trigger = { zeppelin_arctic_region_se_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = se } }

			#1 = { trigger = { zeppelin_arctic_region_var >= 8 zeppelin_arctic_region_pole_mapped_trigger = no } zeppelin_map_arctic_region_effect = { REGION = pole } }
		}
	}
}

### Event Log ###

#Clears all Event log containers
zeppelin_expedition_clear_event_log_containers_je_scope_effect = {
	every_container = {
		limit = {
			has_tag = zeppelin_expedition_event
		}
		destroy_container = yes
	}
	clear_variable_list = zeppelin_expedition_events_list
}

#Creates a new container which is used in the JE to display recent events.
# Execute in the Journal Entry scope.
# An event in this case is a short text which will be displayed in a list.
zeppelin_expedition_event_list_add_container_entry_effect = {
	# ICON = Icon reference, in our case a flag which references a loc key
	# TITLE_LOCALIZATION_KEY = Title localization
	# TOOLTIP_KEY = Localization key for the event description
	# PRIMARY_COUNTRY = Country to reference in event description
	# ADDITIONAL_EFFECTS_CONTAINER = Additional effects in the container scope
	create_container = {
		tags = { mr_mod zeppelin_expedition_event }
		parent = this
		save_scope_as = zeppelin_event_container_scope
	}

	scope:zeppelin_event_container_scope ?= {
		set_variable = {
			name = event_container_icon
			value = $ICON$
		}
		set_variable = {
			name = event_container_primary_country
			value = $PRIMARY_COUNTRY$
		}
		set_variable = {
			name = event_container_title_loc
			value = $TITLE_LOCALIZATION_KEY$
		}
		set_variable = {
			name = event_container_tt
			value = $TOOLTIP_KEY$
		}
		set_variable = {
			name = event_container_date
			value = game_date
		}
		$ADDITIONAL_EFFECTS_CONTAINER$
	}
	add_to_variable_list = {
		name = zeppelin_expedition_events_list
		target = scope:zeppelin_event_container_scope
	}
}
