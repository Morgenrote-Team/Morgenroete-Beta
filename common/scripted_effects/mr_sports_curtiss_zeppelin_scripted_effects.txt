curtiss_set_zeppelin_fuel_hydrogen_effect = {
	custom_tooltip = {
	text = curtiss_set_zeppelin_fuel_hydrogen_effect_tt
		var:curtiss_captain_var ?= {
			set_variable = {
				name = curtiss_zeppelin_fuel_var
				value = flag:curtiss_zeppelin_fuel_hydrogen_flag
			}
		}
		add_modifier = {
			name = curtiss_zeppelin_modification_modifier
			days = 180
		}
		set_variable = {
			name = curtiss_zeppelin_fuel_var
			value = 0
		}
	}
}
