zeppelin_set_zeppelin_fuel_hydrogen_effect = {
	custom_tooltip = {
	text = zeppelin_set_zeppelin_fuel_hydrogen_effect_tt
		var:zeppelin_captain_var ?= {
			set_variable = {
				name = zeppelin_fuel_var
				value = flag:zeppelin_fuel_hydrogen_flag
			}
		}
		add_modifier = {
			name = zeppelin_modification_modifier
			days = 180
		}
		set_variable = {
			name = zeppelin_fuel_var
			value = 0
		}
	}
}
