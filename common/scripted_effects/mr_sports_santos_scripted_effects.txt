santos_ballooning_research_if_record_all_effect = {
	if = {
		limit = {
			OR = {
				santos_altitude_record_global_broken_trigger = yes
				santos_distance_record_global_broken_trigger = yes
				santos_time_record_global_broken_trigger = yes
			}
		}
		big_santos_ballooning_research_effect = yes
	}
	else_if = {
		limit = {
			OR = {
				santos_altitude_record_country_broken_trigger = yes
				santos_distance_record_country_broken_trigger = yes
				santos_time_record_country_broken_trigger = yes
			}
		}
		medium_santos_ballooning_research_effect = yes
	}
	else = {
		small_santos_ballooning_research_effect = yes
	}
}

santos_add_balloonist_adventure_items_effect = {
	### Archaeologist ###
	if = {
		limit = {
			var:santos_balloonist_var = {
				has_trait = lepsius_archaeologist
			}
		}
		random_list = {
			8 = { lepsius_add_x_local_artifacts_effect = { AMOUNT = 2 } }
			2 = { lepsius_add_x_egyptian_artifacts_effect = { AMOUNT = 2 } }
			2 = { lepsius_add_x_mesopotamian_artifacts_effect = { AMOUNT = 2 } }
			2 = { lepsius_add_x_aegean_artifacts_effect = { AMOUNT = 2 } }
			2 = { lepsius_add_x_precolumbian_artifacts_effect = { AMOUNT = 2 } }
		}
	}
	### Ethnologist ###
	if = {
		limit = {
			var:santos_balloonist_var = {
				has_trait = roy_ethnologist
			}
		}
		random_list = {
			1 = { lepsius_add_x_oceanic_artifacts_effect = { AMOUNT = 1 } }
			1 = { lepsius_add_x_fareastern_artifacts_effect = { AMOUNT = 1 } }
		}
	}
	### Naturalist ###
	if = {
		limit = {
			var:santos_balloonist_var = {
				has_trait = ito_naturalist
			}
		}
		dubois_add_x_specimen_effect = { AMOUNT = 2000 }
		dubois_add_x_fossils_effect = { AMOUNT = 2 }
	}
	### Geologist ###
	if = {
		limit = {
			var:santos_balloonist_var = {
				has_trait = agassiz_geologist
			}
		}
		agassiz_add_x_generic_minerals_effect = { AMOUNT = 4000 }
		agassiz_add_x_rare_minerals_effect = { AMOUNT = 1 }
	}
	### Engineer ###
	if = {
		limit = {
			var:santos_balloonist_var = {
				has_trait = engineer
			}
		}
		tesla_aeronautical_engineering_small_add_technology = yes
	}
	### Meteorologist ###
	if = {
		limit = {
			var:santos_balloonist_var = {
				has_trait = curie_meteorological_physicist
			}
		}
		rapanui_add_x_oceanographic_understanding_effect = { AMOUNT = 3 }
		curie_add_x_physical_data_effect = { AMOUNT = 5 }
		curie_add_x_magnetic_data_effect = { AMOUNT = 10 }
	}
}

santos_new_balloon_altitude_effect = {
	if = {
		limit = {
			has_variable = santos_balloonist_var
		}
		set_variable = {
			name = new_balloon_altitude_var
			value = santos_new_balloon_altitude_value #Arthur Berson und Reinhard Süring did 1901 max. 10.800; value may be higher, but should not give access to the stratosphere
		}
		var:santos_balloonist_var ?= {
			set_variable = {
				name = santos_balloon_altitude_record_personal_var
				value = root.var:new_balloon_altitude_var
			}
		}
	}
}

santos_balloon_country_altitude_record_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = santos_balloon_altitude_record_var
			}
		}
		set_variable = {
			name = santos_balloon_altitude_record_var
			value = var:new_balloon_altitude_var
		}
	}
	balloonist_experience_gain_effect = { #if no record is broken, a total of 50 experience is added
		AMOUNT = 10
	}
	if = {
		limit = {
			var:new_balloon_altitude_var > var:santos_balloon_altitude_record_var
		}
		set_variable = {
			name = santos_balloon_altitude_record_var
			value = var:new_balloon_altitude_var
		}
		balloonist_experience_gain_effect = { #if country record is broken, a total of 25 experience is added
			AMOUNT = 15
		}
	}
	if = {
		limit = {
			var:new_balloon_altitude_var > global_var:santos_balloon_altitude_record_global_var
		}
		set_global_variable = {
			name = santos_balloon_altitude_record_global_var
			value = var:new_balloon_altitude_var
		}
		set_global_variable = {
			name = santos_balloon_altitude_record_holder_global_var
			value = root
		}
		save_scope_as = santos_record_breaking_country

		if = {
			limit = { NOT = { has_modifier = santos_balloon_altitude_record_modifier } }
			every_country = { #removes modifier from old country
				limit = {
					has_modifier = santos_balloon_altitude_record_modifier
				}
				remove_modifier = santos_balloon_altitude_record_modifier
				post_notification = santos_altitude_record_broken_notification
			}
			add_modifier = { #adds modifier to new country
				name = santos_balloon_altitude_record_modifier
				multiplier = var:new_balloon_altitude_var ##1 Prestige for every 500 Meters (multiplies with 0.002 Prestige)
			}
		}

santos_gordon_bennett_cup_effect = {
	remove_global_variable = santos_gordon_bennett_cup_winner_global_var
	every_country = {
		limit = {
			has_modifier = santos_gordon_bennett_cup_modifier
		}
		remove_modifier = santos_gordon_bennett_cup_modifier
	}

	ordered_country = {
		limit = {
			santos_has_balloonist_trigger = yes
		}
		order_by = santos_gordon_bennett_cup_score
		max = 1 #Limits the list to top country
		check_range_bounds = no #prevents errors

		#This part checks one after another whether the first, second, etc. ranked countries are saved already.
		#Luckily the ordered_country effect iterates through the countries one after another.
		#So the first country gets saved to the first variable but as that variable now is set,
		# the second country gets saved to the second variable and so on.
		if = {
			limit = {
				NOT = { has_global_variable = santos_gordon_bennett_cup_winner_global_var }
			}
			set_global_variable = {
				name = santos_gordon_bennett_cup_winner_global_var
				value = this
			}
			balloonist_experience_gain_effect = { AMOUNT = 25 }
			add_modifier = {
				name = santos_gordon_bennett_cup_modifier
			}
			save_scope_as = santos_gordon_bennett_cup_winner
			post_notification = santos_gordon_bennett_cup_notification
		}
	}
}
