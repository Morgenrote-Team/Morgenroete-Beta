#Effect to set up a container for a experiment and to add it to the global variable list
curie_create_experiment_container_and_add_to_global_variable_list_effect = {
	# NAME = a localized flag (use Localize(ScriptContainer.GetName) to access the name loc)
	# ON_CREATED = effect in the on_created
	# LIST_NAME = name of the list
	create_container = {
		name = $NAME$
		tags = { mr_mod curie_experiments }
		on_created = {
			$ON_CREATED$
		}
		save_scope_as = container_scope
	}

	add_to_global_variable_list = {
		name = curie_experiments_global_list
		target = scope:container_scope
	}
	add_to_global_variable_list = {
		name = $LIST_NAME$
		target = scope:container_scope
	}
}

	# This is going to be a long list of ifs.

	# Don't forget to add 1 to the respective counter (e.g. curie_discovered_experiments_global_number_var)
	# For the Discovery, don't forget to add the year as a variable.
	# If possible add the most fitting country's definition to the variable.

	set_global_variable = {
		name = curie_discovered_experiments_global_number_var

	#Voltaic pile, Young 1801, Great Britain

		limit = {
			game_date >= 1801.1.1
		}

		#Discovered
		curie_create_experiment_container_and_add_to_global_variable_list_effect = {
			NAME = CURIE_EXPERIMENT_YOUNG_NAME
			LIST_NAME = curie_experiments_global_list
			ON_CREATED = "
				set_variable = {
					name = name
					value = flag:CURIE_EXPERIMENT_YOUNG_NAME
				}
				set_variable = {
					name = tooltip
					value = flag:CURIE_EXPERIMENT_YOUNG_TT
				}
				set_variable = {
					name = curie_experiment_number_var
					value = 1
				}
				set_variable = {
					name = year
					value = 1801
				}

					name = discovering_country
					value = cd:GBR
				}

					name = discoverer
					value = flag:curie_thomas_young_loc
