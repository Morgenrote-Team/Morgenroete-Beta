#Effect to set up a container for a crossing and to add it to the global variable list
santos_create_crossing_container_and_add_to_global_variable_list_effect = {
	# NAME = a localized flag (use Localize(ScriptContainer.GetName) to access the name loc)
	# ON_CREATED = effect in the on_created
	# LIST_NAME = name of the list
	create_container = {
		name = $NAME$
		tags = { mr_mod santos_crossings }
		on_created = {
			$ON_CREATED$
		}
		save_scope_as = container_scope
	}

	add_to_global_variable_list = {
		name = santos_balloon_crossings_global_list
		target = scope:container_scope
	}
	add_to_global_variable_list = {
		name = $LIST_NAME$
		target = scope:container_scope
	}
}
