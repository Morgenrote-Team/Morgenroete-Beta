#Effect to add $AMOUNT$ Magnetic Data
curie_add_x_magnetic_data_effect = {
	# Amount = amount of Magnetic Data to add
	save_temporary_scope_value_as = { #This is used to display the amount in the tooltip
		name = magnetic_data_amount
		value = $AMOUNT$
	}

	#This part is used to display an accurate tooltip (Gain, loss, etc.)
	if = { #Negative Plural Tooltip if $AMOUNT$ is negative
		limit = { $AMOUNT$ < -1 }
		custom_tooltip = curie_add_x_magnetic_data_minus_x_tt
	}
	if = { #Negative Singular Tooltip if $AMOUNT$ is negative
		limit = { $AMOUNT$ = -1 }
		custom_tooltip = curie_add_x_magnetic_data_minus_1_tt
	}
	if = { #Neutral Tooltip if $AMOUNT$ is 0
		limit = { $AMOUNT$ = 0 }
		custom_tooltip = curie_add_x_magnetic_data_0_tt
	}
	if = { #Positive Plural Tooltip if $AMOUNT$ is positive
		limit = { $AMOUNT$ = 1 }
		custom_tooltip = curie_add_x_magnetic_data_plus_1_tt
	}
	if = { #Positive Plural Tooltip if $AMOUNT$ is positive
		limit = { $AMOUNT$ > 1 }
		custom_tooltip = curie_add_x_magnetic_data_plus_x_tt
	}

	#This part adds the AMOUNT of National Magnetic Data
	if = { #Fallback if the variable does not exist yet
		limit = {
			NOT = {
				has_variable = curie_magnetic_data_var
			}
		}
		set_variable = {
			name = curie_magnetic_data_var
			value = 0
		}
	}
	change_variable = {
		name = curie_magnetic_data_var
		add = $AMOUNT$
	}

	#This part adds the AMOUNT of Global Magnetic Data
	if = { #Fallback if the variable does not exist yet
		limit = {
			NOT = {
				has_global_variable = curie_total_magnetic_data_global_var
			}
		}
		set_global_variable = {
			name = curie_total_magnetic_data_global_var
			value = 0
		}
	}
	change_global_variable = {
		name = curie_total_magnetic_data_global_var
		add = $AMOUNT$
	}
}

curie_set_magnetic_association_effect = {
	curie_add_magnetic_association_member = yes
	set_variable = curie_magnetic_station_var

	country_definition = {
		set_global_variable = {
			name =  curie_magnetic_association_global_var
			value = THIS
		}
	}

	random_scope_state = {
		limit = {
			has_building = building_verrier_observatory
		}
		state_region = {
			set_global_variable = { #This is used in the GUI to display the state
            	name = curie_magnetic_association_city_global_var
           		value = this
        	}
		}
	}

	custom_tooltip = curie_set_magnetic_association_effect_tt
}

		name = curie_magnetic_association_members_global_var
		add = 1
	}
	set_variable = curie_magnetic_association_member_var

	custom_tooltip = curie_add_magnetic_association_member_tt
}

curie_magnetic_progress_gain_effect = {
	if = {
		limit = {
			NOT = { 
				has_global_variable = curie_total_magnetic_data_global_var
			}
		}
		set_global_variable = {
			name = curie_total_magnetic_data_global_var
			value = 0
		}
	}
	change_global_variable = {
		name = curie_total_magnetic_data_global_var
		add = {
			value = global_var:curie_magnetic_association_members_global_var
		}
	}
	if = {
		limit = {
			var:curie_total_magnetic_data_global_var >= curie_magnetic_progress_goal
		}
		curie_magnetic_research_effect = yes
	}
}

curie_magnetic_research_effect = {
	change_global_variable = {
		name = curie_total_magnetic_data_global_var
		subtract = curie_magnetic_progress_goal
	}
	change_global_variable = {
		name = curie_magnetic_research_global_var
		add = 1
	}

	agassiz_add_x_geological_knowledge_effect = { AMOUNT = 1 }

		limit = {
			has_variable = curie_magnetic_association_member_var
		}
		trigger_event = {
			id = curie.103
		}
	}
}

#Effect to add $AMOUNT$ Physical Data
curie_add_x_physical_data_effect = {
	# Amount = amount of physical_data to add
	save_temporary_scope_value_as = { #This is used to display the amount in the tooltip
		name = physical_data_amount
		value = $AMOUNT$
	}

	#This part is used to display an accurate tooltip (Gain, loss, etc.)
	if = { #Negative Plural Tooltip if $AMOUNT$ is negative
		limit = { $AMOUNT$ < -1 }
		custom_tooltip = curie_add_x_physical_data_minus_x_tt
	}
	if = { #Negative Singular Tooltip if $AMOUNT$ is negative
		limit = { $AMOUNT$ = -1 }
		custom_tooltip = curie_add_x_physical_data_minus_1_tt
	}
	if = { #Neutral Tooltip if $AMOUNT$ is 0
		limit = { $AMOUNT$ = 0 }
		custom_tooltip = curie_add_x_physical_data_0_tt
	}
	if = { #Positive Plural Tooltip if $AMOUNT$ is positive
		limit = { $AMOUNT$ = 1 }
		custom_tooltip = curie_add_x_physical_data_plus_1_tt
	}
	if = { #Positive Plural Tooltip if $AMOUNT$ is positive
		limit = { $AMOUNT$ > 1 }
		custom_tooltip = curie_add_x_physical_data_plus_x_tt
	}

	#Yearly physical_data are used for the botanical report, they are added with each normal physical_data
	if = { #Fallback if the variable does not exist yet
		limit = {
			NOT = {
				has_variable = curie_yearly_physical_data_var
			}
		}
		set_variable = {
			name = curie_yearly_physical_data_var
			value = 0
		}
	}
	change_variable = {
		name = curie_yearly_physical_data_var
		add = $AMOUNT$
	}

	#This part adds the AMOUNT of physical_data
	if = { #Fallback if the variable does not exist yet
		limit = {
			NOT = {
				has_variable = curie_physical_data_var
			}
		}
		set_variable = {
			name = curie_physical_data_var
			value = 0
		}
	}
	change_variable = {
		name = curie_physical_data_var
		add = scope:physical_data_amount
	}

	#curie_physicist_total_physical_data_var is used to keep track of all the physical_data that a specific physicist found.
	#It is used to show how successful the physicist is.
	#Every time a normal physicist object is added, the same number gets added here.
	if = {
		limit = { #Only works if there actually is a physicist
			has_variable = curie_physicist_var
		}
		var:curie_physicist_var = {
			if = { #Fallback if the variable does not exist yet
				limit = {
					NOT = {
						has_variable = curie_physicist_total_physical_data_var
					}
				}
				set_variable = {
					name = curie_physicist_total_physical_data_var
					value = 0
				}
			}
			change_variable = { #The amount of physical_data that this physicist discovered
				name = curie_physicist_total_physical_data_var
				add = scope:physical_data_amount
			}
		}
	}
}
