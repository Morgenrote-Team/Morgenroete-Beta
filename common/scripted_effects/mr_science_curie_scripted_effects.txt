#Effect to add $AMOUNT$ Physical Data
curie_add_x_physical_data_effect = {
	# Amount = amount of physical_data to add
	save_temporary_scope_value_as = { #This is used to display the amount in the tooltip
		name = physical_data_amount
		value = $AMOUNT$
	}

	#This part is used to display an accurate tooltip (Gain, loss, etc.)
	if = { #Negative Plural Tooltip if $AMOUNT$ is negative
		limit = { $AMOUNT$ < -1 }
		custom_tooltip = curie_add_x_physical_data_minus_x_tt
	}
	if = { #Negative Singular Tooltip if $AMOUNT$ is negative
		limit = { $AMOUNT$ = -1 }
		custom_tooltip = curie_add_x_physical_data_minus_1_tt
	}
	if = { #Neutral Tooltip if $AMOUNT$ is 0
		limit = { $AMOUNT$ = 0 }
		custom_tooltip = curie_add_x_physical_data_0_tt
	}
	if = { #Positive Plural Tooltip if $AMOUNT$ is positive
		limit = { $AMOUNT$ = 1 }
		custom_tooltip = curie_add_x_physical_data_plus_1_tt
	}
	if = { #Positive Plural Tooltip if $AMOUNT$ is positive
		limit = { $AMOUNT$ > 1 }
		custom_tooltip = curie_add_x_physical_data_plus_x_tt
	}

	#Yearly physical_data are used for the botanical report, they are added with each normal physical_data
	if = { #Fallback if the variable does not exist yet
		limit = {
			NOT = {
				has_variable = curie_yearly_physical_data_var
			}
		}
		set_variable = {
			name = curie_yearly_physical_data_var
			value = 0
		}
	}
	change_variable = {
		name = curie_yearly_physical_data_var
		add = $AMOUNT$
	}

	#This part adds the AMOUNT of physical_data
	if = { #Fallback if the variable does not exist yet
		limit = {
			NOT = {
				has_variable = curie_physical_data_var
			}
		}
		set_variable = {
			name = curie_physical_data_var
			value = 0
		}
	}
	change_variable = {
		name = curie_physical_data_var
		add = scope:physical_data_amount
	}

	#curie_physicist_total_physical_data_var is used to keep track of all the physical_data that a specific physicist found.
	#It is used to show how successful the physicist is.
	#Every time a normal physicist object is added, the same number gets added here.
	if = {
		limit = { #Only works if there actually is a physicist
			has_variable = curie_physicist_var
		}
		var:curie_physicist_var = {
			if = { #Fallback if the variable does not exist yet
				limit = {
					NOT = {
						has_variable = curie_physicist_total_physical_data_var
					}
				}
				set_variable = {
					name = curie_physicist_total_physical_data_var
					value = 0
				}
			}
			change_variable = { #The amount of physical_data that this physicist discovered
				name = curie_physicist_total_physical_data_var
				add = scope:physical_data_amount
			}
		}
	}
}

#Effect to add a specific searched formula
curie_add_searched_formula_effect = {
	#NUMBER --> Number of the Formula in the List

	save_temporary_scope_value_as = { #This is used to display the amount in the tooltip
		name = value
		value = $NUMBER$
	}

	#Actual Effect
	if = {
		limit = {
			NOT = {
				any_in_global_list = {
					variable = curie_formulas_global_list
					var:curie_formula_number_var ?= $NUMBER$
					has_variable = curie_formula_discovered_var
				}
			}
		}
		set_variable = {
			name = curie_searched_variable
			value = $NUMBER$
		}
	}
}

#Effect to add the specific formula which a physicist searched as a discovered formula
curie_add_discovered_formula_effect = {
	#NUMBER --> Number of the Formula

	save_temporary_scope_value_as = { #This is used to display the amount in the tooltip
		name = value
		value = $NUMBER$
	}

	random_in_global_list = {
		variable = curie_formulas_global_list

		limit = {
			var:curie_formula_number_var ?= $NUMBER$
		}


		set_variable = {
			name = year
			value = year
		}
		set_variable = {
			name = discovering_country
			value = prev.country_definition
		}
		if = {
			limit = {
				prev.var:curie_physicist_var ?= {
					has_variable = com_name_full_no_formatting
				}
			}
			set_variable = {
				name = discoverer
				value = prev.var:curie_physicist_var.var:com_name_full_no_formatting
			}
		}
		set_variable = curie_formula_discovered_var
	}
	change_global_variable = {
		name = curie_discovered_formulas_global_number_var
		add = 1
	}
	mr_change_variable_min_0_effect = { AMOUNT = 1 NAME = curie_discovered_formulas_country_number_var }
	var:curie_physicist_var ?= {
		mr_change_variable_min_0_effect = { AMOUNT = 1 NAME = curie_discovered_formulas_character_number_var }
	}
}
