#Effect to add discovered formulas in 1836 [MORGENRÖTE]
curie_set_formulas_history_effect = {
	### Tutorial ###
	# This is going to be a long list of ifs.
	# Each if checks whether it has been a certain year or date.
	# If the date of the discovery  has passed, the element will be added to the correct list.
	# Don't forget to add 1 to the respective counter (e.g. curie_discovered_formulas_global_number_var)
	# For the Discovery, don't forget to add the year as a variable.
	# If possible add the most fitting country's definition to the variable.

	#This part sets up the global base value
	set_global_variable = {
		name = curie_discovered_formulas_global_number_var
		value = 0
	}
	##############################################################################################

	#########################
	# PRE 1836 FORMULAS #
	#########################

	##############################################################################################

	#Complex Analysis, Gauß 1810, Hanover
	if = {
		limit = {
			game_date >= 1810.1.1
		}

		#Dicovered
		add_to_global_variable_list = {
			name = curie_discovered_formulas_global_list
			target = 1
		}
		change_global_variable = {
			name = curie_discovered_formulas_global_number_var
			add = 1
		}

		cd:HAN = { #This scopes to the country definition. E.g cd:HAM = Hamburg
			set_global_variable = {
				name = curie_discovered_formula_1_global_var
				value = this
			}
		}

		set_global_variable = {
			name = curie_discovered_formula_1_year_global_var
			value = 1810
		}
	}
	##############################################################################################

	#Differential Equations, Fourier 1822, France
	if = {
		limit = {
			game_date >= 1822.1.1
		}

		#Discovered
		add_to_global_variable_list = {
			name = curie_discovered_formulas_global_list
			target = 2
		}
		change_global_variable = {
			name = curie_discovered_formulas_global_number_var
			add = 1
		}

		cd:FRA = { #This scopes to the country definition. E.g cd:HAM = Hamburg
			set_global_variable = {

			}

		}
	}
	##############################################################################################

	#Fourier Analysis, Fourier 1822, France
	if = {
		limit = {
			game_date >= 1822.1.1
		}

		#Discovered
		add_to_global_variable_list = {
			name = curie_discovered_formulas_global_list
			target = 3
		}
		change_global_variable = {
			name = curie_discovered_formulas_global_number_var
			add = 1
		}

		cd:FRA = { #This scopes to the country definition. E.g cd:HAM = Hamburg
			set_global_variable = {
				name = curie_discovered_formula_3_global_var
				value = this
			}
		}

		set_global_variable = {
			name = curie_discovered_formula_3_year_global_var
			value = 1822
		}
	}
	##############################################################################################

	#Non Euclidian Geometry, Lobachevsky 1829, Russia
	if = {
		limit = {
			game_date >= 1829.1.1
		}

		#Discovered
		add_to_global_variable_list = {
			name = curie_discovered_formulas_global_list
			target = 4
		}
		change_global_variable = {
			name = curie_discovered_formulas_global_number_var
			add = 1
		}

		cd:RUS = { #This scopes to the country definition. E.g cd:HAM = Hamburg

			value = 1829

	#Group Theory, Galois 1830, France
	if = {
		limit = {
			game_date >= 1830.1.1
		}

		#Discovered
		add_to_global_variable_list = {
			name = curie_discovered_formulas_global_list
			target = 5

		change_global_variable = {
			name = curie_discovered_formulas_global_number_var
