#Checks if the scope is an AI and then starts Physics Projects
ai_start_physics_project_for_ai_effect = {
	if = {
		limit = {
			curie_has_active_physicist_no_tooltip_trigger = yes
			tesla_is_engineering_automated_trigger = yes
		}
		ai_physics_automation_effect = yes
	}
}

#Handles Physics for the AI and player automation
ai_physics_automation_effect = {
	hidden_effect = {
		if = {
			limit = {
				curie_is_physics_automated_trigger = yes
				curie_has_active_physicist_no_tooltip_trigger = yes
			}

			### Physics Experiment ###

			#Uses a random list to decide which project to do.
			# This allows for weighting using various factors.
			random_list = {
				### Physics Experiment ###
				300 = { #High Physical Priority
					#Triggers
					trigger = {
						NOT = {
							has_journal_entry = je_curie_physics_experiment
						}
						NOR = {
							has_modifier = curie_physics_experiment_successful_modifier
							has_modifier = curie_physics_experiment_failure_modifier
						}
					}

					#Modifiers / Weighting
					modifier = {
						if = {
							limit = {
								curie_experiment_can_discover_trigger = no #If a Physicist can not do a historical experiment, this has low priority
							}
							add = -1000
						}
						if = {
							limit = {
								var:curie_physicist_var ?= {
									has_trait = cautious
								}
							}
							add = 100
						}
						if = {
							limit = {
								var:curie_physicist_var ?= {
									has_trait = meticulous
								}
							}
							add = 100
						}
					}

					#Effect
					curie_physics_experiment_button_effect = yes
				}
				### New Formula ###
				200 = { #High Mathematical Priority
					#Triggers
					trigger = {
						curie_has_mathematical_physicist_trigger = yes
						NOT = {
							has_journal_entry = je_curie_formula_discovery
						}
						curie_formula_can_discover_trigger = yes
						var:curie_physical_data_var >= curie_formula_cost_value
					}

					#Modifiers / Weighting
					modifier = {
						if = {
							limit = {
								var:curie_physicist_var ?= {
									has_trait = innovative
								}
							}
							add = 100
						}
						if = {
							limit = {
								var:curie_physicist_var ?= {
									has_trait = erudite
								}
							}
							add = 50
						}
					}

					#Effect
					curie_formula_discovery_button_effect = yes
				}
				### Improve Mathematics ###
				100 = { #Mid Mathematical Priority
					#Triggers
					trigger = {
						curie_has_mathematical_physicist_trigger = yes
						NOT = {
							has_journal_entry = je_curie_improve_mathematics_project
						}
						NOT = {
							has_variable = curie_improve_mathematics_project_building_choser_var
						}
						curie_country_has_building_valid_for_mathematics_project_trigger = yes
					}

					#Modifiers / Weighting
					modifier = {
						if = {
							limit = {
								var:curie_physicist_var ?= {
									has_trait = innovative
								}
							}
							add = 100
						}
						if = {
							limit = {
								var:curie_physicist_var ?= {
									has_trait = erudite
								}
							}
							add = 50
						}
					}

					#Effect
					curie_physicist_improve_mathematics_effect = yes
				}
			}
		}
	}
}
