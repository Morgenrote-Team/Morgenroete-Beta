#This effect replaces the current Income Modifier
curtiss_update_tourism_income_effect = {
	if = {
		limit = {
			has_technology_researched = curtiss_tourism_tech
		}
		hidden_effect = {
			remove_modifier = curtiss_tourism_income_modifier
		}
		add_modifier = {
			name = curtiss_tourism_income_modifier
			multiplier = curtiss_money_amount_multiplier_tourism
		}
	}
}

#Effect to update level modifier for an airline
curtiss_update_airline_effect = {
	if = {
		limit = {
			var:curtiss_flight_route >= 10
			NOT = {
				has_modifier = curtiss_airline_level_6_modifier
			}
		}
		add_modifier = curtiss_airline_level_6_modifier
		if = {
			limit = { has_modifier = curtiss_airline_level_1_modifier }
			remove_modifier = curtiss_airline_level_1_modifier
		}
		if = {
			limit = { has_modifier = curtiss_airline_level_2_modifier }
			remove_modifier = curtiss_airline_level_2_modifier
		}
		if = {
			limit = { has_modifier = curtiss_airline_level_3_modifier }
			remove_modifier = curtiss_airline_level_3_modifier
		}
		if = {
			limit = { has_modifier = curtiss_airline_level_4_modifier }
			remove_modifier = curtiss_airline_level_4_modifier
		}
		if = {
			limit = { has_modifier = curtiss_airline_level_5_modifier }
			remove_modifier = curtiss_airline_level_5_modifier
		}
	}
	else_if = {
		limit = {
			var:curtiss_flight_route >= 8
			NOT = {
				has_modifier = curtiss_airline_level_5_modifier
				var:curtiss_flight_route >= 10
			}

curtiss_add_pilot_num_kills_effect = {
	var:curtiss_pilot_var ?= {
		if = { #Fallback if the variable does not exist yet
			limit = {
				NOT = {
					has_variable = curtiss_pilot_num_kills_var
				}
			}
			set_variable = {
				name = curtiss_pilot_num_kills_var
				value = 0
			}
		}
		change_variable = {
			name = curtiss_pilot_num_kills_var
			add = $AMOUNT$
		}
	}
}

curtiss_monthly_pilot_num_kills_effect = {
	#Basic
	random_list = {
		1 = {}
		1 = { curtiss_add_pilot_num_kills_effect = { AMOUNT = 1 } pilot_experience_gain_effect = { AMOUNT = 5 } }
		1 = { curtiss_add_pilot_num_kills_effect = { AMOUNT = 2 } pilot_experience_gain_effect = { AMOUNT = 10 } }
		1 = { curtiss_add_pilot_num_kills_effect = { AMOUNT = 3 } pilot_experience_gain_effect = { AMOUNT = 15 } }
	}
	#Experience
	if = {
		limit = {
			var:curtiss_pilot_var = {
				var:curtiss_pilot_experience >= 100
			}
		}
		curtiss_add_pilot_num_kills_effect = { AMOUNT = 1 }
		pilot_experience_gain_effect = { AMOUNT = 5 }
	}
	#Traits
	if = {
		limit = {
			var:curtiss_pilot_var = {
				has_trait = curtiss_pilot_fighter
			}
		}
		curtiss_add_pilot_num_kills_effect = { AMOUNT = 1 }
		pilot_experience_gain_effect = { AMOUNT = 5 }
	}
	if = {
		limit = {
			var:curtiss_pilot_var = {
				has_trait = curtiss_pilot_ace
			}
		}
		curtiss_add_pilot_num_kills_effect = { AMOUNT = 1 }
		pilot_experience_gain_effect = { AMOUNT = 5 }
	}
	#Vices
	if = {
		limit = {
			var:curtiss_pilot_var = {
				OR = {
					has_trait = alcoholic
					has_trait = opium_addiction
					has_trait = cocaine_addiction
					has_trait = shellshocked
					has_trait = psychological_affliction
				}
			}
		}
		curtiss_add_pilot_num_kills_effect = { AMOUNT = -1 }
		pilot_experience_gain_effect = { AMOUNT = -5 }
	}
}

#Effect that gets fired when a country has 30 Schneider Trophy points
curtiss_schneider_trophy_victory_effect = {
	if = {
		limit = {
			var:curtiss_schneider_trophy_point_var >= 30
		}

		#Schneider Trophy History log
		curtiss_schneider_trophy_add_history_container_effect = { PLANE_MODEL = flag:mr_empty_flag }

		#Victory Event
		trigger_event = {
			id = curtiss.111
			popup = yes
		}
		#Resetting Schneider Trophy Points
		set_variable = {
			name = curtiss_schneider_trophy_point_var
			value = 0
		}

		#New Host State
		ordered_scope_state = {
			limit = {
				is_incorporated = yes
				any_scope_building = {
					is_building_type = building_port
				}
				any_scope_building = {
					is_building_type = building_urban_center
					level >= 10
				}
			}
			order_by = total_urbanization
			max = 1 #Limits to the state with highest urbanization

			save_scope_as = curtiss_schneider_trophy_state
		}
		if = {
			limit = { NOT = { exists = scope:curtiss_schneider_trophy_state } }
			ordered_scope_state = {
				limit = {
					is_coastal = yes
				}
				order_by = total_urbanization
				max = 1 #Limits to the state with highest urbanization

				save_scope_as = curtiss_schneider_trophy_state
			}
		}
		if = {
			limit = { NOT = { exists = scope:curtiss_schneider_trophy_state } }
			capital = {
				save_scope_as = curtiss_schneider_trophy_state
			}
		}

		#Ends the Race
		je:je_curtiss_schneider_trophy_race_contextless = {
			remove_variable = curtiss_schneider_trophy_race_active_var

			#Sets new Host variables
			set_variable = {
				name = curtiss_schneider_trophy_host_nation_var
				value = prev
			}
			set_variable = {
				name = curtiss_schneider_trophy_host_state_var
				value = scope:curtiss_schneider_trophy_state
			}
			set_variable = {
				name = curtiss_schneider_trophy_host_state_region_var
				value = scope:curtiss_schneider_trophy_state.state_region
			}

			#Resets Event Log
			curtiss_schneider_trophy_clear_event_log_containers_je_scope_effect = yes

			#Informs other countries about winner and removes variables, etc.
			every_scope_je_involved = {
				limit = {
					NOT = { je:je_curtiss_schneider_trophy_race_contextless.var:curtiss_schneider_trophy_host_nation_var ?= this }
				}

				#Resets Schneider Trophy Points
				set_variable = {
					name = curtiss_schneider_trophy_point_var
					value = 0
				}

				#Sends Event
				trigger_event = curtiss.112

				#Resets consecutive Victories
				if = {
					limit = { has_variable = curtiss_schneider_trophy_consecutive_victories_var }
					set_variable = {
						name = curtiss_schneider_trophy_consecutive_victories_var
						value = 0
					}
				}
				if = {
					limit = {
						has_modifier = curtiss_schneider_trophy_1_win_modifier
					}
					remove_modifier = curtiss_schneider_trophy_1_win_modifier
				}
				if = {
					limit = {
						has_modifier = curtiss_schneider_trophy_2_win_modifier
					}
					remove_modifier = curtiss_schneider_trophy_2_win_modifier
				}
				#Adds Schneider Veteran Trait to Pilot and adds experience
				if = {
					limit = {
						NOT = {
							var:curtiss_pilot_var ?= {
								has_trait = curtiss_schneider_veteran
							}
						}
					}
					var:curtiss_pilot_var ?= {
						add_trait = curtiss_schneider_veteran
					}
				}
				curtiss_add_pilot_num_schneider_races_effect = { AMOUNT = 1 }
				pilot_experience_gain_effect = {
					AMOUNT = 25
				}
				#Resets other Variables
				remove_modifier = curtiss_schneider_trophy_support_modifier
				var:curtiss_pilot_var ?= {
					set_character_busy_and_immortal = no
					remove_modifier = curtiss_schneider_pilot_modifier
				}
			}
		}
	}
}

	}

curtiss_airplane_country_speed_record_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = curtiss_airplane_speed_record_var
			}
		}
		set_variable = {
			name = curtiss_airplane_speed_record_var
			value = var:new_airplane_speed_var
		}
	}
	pilot_experience_gain_effect = { #if no record is broken, a total of 10 experience is added
		AMOUNT = 10
	}
	aeronautical_engineer_experience_gain_effect = {
		AMOUNT = 10
	}
	if = {
		limit = {
			var:new_airplane_speed_var > var:curtiss_airplane_speed_record_var
		}
		set_variable = {
			name = curtiss_airplane_speed_record_var
			value = var:new_airplane_speed_var
		}
		pilot_experience_gain_effect = { #if country record is broken, a total of 25 experience is added
			AMOUNT = 25
		}
		aeronautical_engineer_experience_gain_effect = {
			AMOUNT = 25
		}
	}
	if = {
		limit = {
			var:new_airplane_speed_var > global_var:curtiss_airplane_speed_record_global_var
		}
		set_global_variable = {
			name = curtiss_airplane_speed_record_global_var
			value = var:new_airplane_speed_var
		}
		set_global_variable = {
			name = curtiss_airplane_speed_record_holder_global_var
			value = root
		}
		save_scope_as = curtiss_record_breaking_country

		if = {
			limit = { NOT = { has_modifier = curtiss_airplane_speed_record_modifier } }
			every_country = { #removes modifier from old country
				limit = {
					has_modifier = curtiss_airplane_speed_record_modifier
				}
				post_notification = curtiss_speed_record_broken_notification
				remove_modifier = curtiss_airplane_speed_record_modifier
			}
			add_modifier = { #adds modifier to new country
				name = curtiss_airplane_speed_record_modifier
			}
		}

		pilot_experience_gain_effect = { #if world record is broken, a total of 50 experience is added
			AMOUNT = 50
		}
		aeronautical_engineer_experience_gain_effect = {
			AMOUNT = 50
		}
	}
}
