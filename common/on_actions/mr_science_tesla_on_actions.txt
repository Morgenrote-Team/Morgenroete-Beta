on_yearly_pulse_country = {
	on_actions = {
		tesla_on_yearly_pulse_country
	}
}

tesla_on_yearly_pulse_country = {
	events = {
		tesla.102 #Tesla 102: Return of a Prince!
	}
}

on_acquired_technology = {
	on_actions = {
		tesla_on_acquired_technology
	}
}

tesla_on_acquired_technology = {
	events = {
		gaudi.1005 #Gaudí 1005: Invention of the Electric Telegraph (Morse)
		gaudi.1006 #Gaudí 1006: The first Automobile with an Internal Combustion Engine
	}
}

on_building_built = {
	on_actions = {
		tesla_on_building_built
	}
}

tesla_on_building_built = {
	effect = {
		if = {
			limit = {
				owner = {
					any_scope_building = {
						is_building_type = building_tesla_dam
						level = 1
					}
				}
			}
			owner = {
				trigger_event = {
					id = tesla.5000 #Lepsius 5000: Mega Dam
				}
			}
		}
	}
}
