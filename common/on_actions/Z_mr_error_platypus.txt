# Fires after everyone enters the game from the lobby for the first time for each campaign
# At this point we DO know who the players are
# No root scope set
on_game_started_after_lobby = {
	on_actions = {
		mr_on_game_started_after_lobby_platypus
	}
}

mr_on_game_started_after_lobby_platypus = {
	effect = {
		random_country = {
			### Adds the error Platypus ###
			com_add_custom_error_deer_icon = {
				icon = mr_error_playtypus
			}
		}
	}
}
