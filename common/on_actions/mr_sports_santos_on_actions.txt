on_half_yearly_pulse_country = {
	on_actions = {
		delay = { days = { 1 180 } } #Learned this from Dementive
		santos_balloons_on_half_yearly_pulse_country
	}
}

on_monthly_pulse = {
	on_actions = {
		santos_on_monthly_pulse
	}
}

santos_on_yearly_pulse_country = {
	effect = {
		curtiss_update_tourism_income_effect = yes
		if = {
			limit = {
				NOT = {
					has_global_variable = santos_gordon_bennet_cup_cooldown_global_var
				}
				has_modifier = santos_gordon_bennett_cup_modifier
			}
			set_global_variable = {
				name = santos_gordon_bennet_cup_cooldown_global_var
				days = 500
			}
			santos_gordon_bennett_cup_effect = yes
		}
	}
	events = {

#		curtiss.602 #Curtiss 602: First crossing of the Alps in a balloon by Francisque Arban
#		curtiss.603 #Curtiss 603: Francisque Arban disappeared during a balloon flight over the Mediterranean Sea
#		curtiss.609 #Curtiss 609: Nadar takes first aerial photos from a balloon

#		curtiss.611 #Curtiss 611: Glaisher and Coxwell set record for highest balloon flight

	}

santos_balloons_on_half_yearly_pulse_country = {
	random_events = {
		500 = 0
		1000 = santos.601 # Balloonist tries to set new Altitude record
		1000 = santos.602 # Balloonist tries to set new Distance record
		1000 = santos.603 # Balloonist tries to set new Time record
		200 = santos.670 # Balloonist died in accident
		200 = santos.675 # Balloonist died in accident (hidden)
		1 = santos.673 # Balloonist killed by comet
		1 = santos.674 # Death by Whale
	}
}

santos_on_monthly_pulse = {
	effect = {
		if = {
			limit = {
				NOT = {
					has_variable = santos_balloon_fuel_var
				}
			}
			set_variable = {
				name = santos_balloon_fuel_var
				value = 6
			}
		}
		if = {
			limit = {
				var:santos_balloon_fuel_var < 6
			}
			change_variable = {
				name = santos_balloon_fuel_var
				add = 1
			}
		}
	}
}
