character_is_artist = {
    base = scope_character
    conditions = {
        OR = {
            has_role = character_role_gaudi_architect
            has_role = character_role_elgar_composer
            has_role = character_role_manzoni_writer
            has_role = character_role_klimt_painter
        }
    }
}

character_is_architect = {
    base = scope_character
    conditions = {
        has_role = character_role_gaudi_architect
    }
}

character_is_composer = {
    base = scope_character
    conditions = {
        has_role = character_role_elgar_composer
    }
}

character_is_writer = {
    base = scope_character
    conditions = {
        has_role = character_role_manzoni_writer
    }
}

character_is_local_writer = {
    base = scope_character
    conditions = {
        has_role = character_role_manzoni_writer
        NOT = { owner.var:manzoni_writer_var ?= this }
    }
}


character_is_painter = {
    base = scope_character
    conditions = {
        has_role = character_role_klimt_painter
    }
}
