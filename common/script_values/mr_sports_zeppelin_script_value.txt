#Zeppelin Speed
zeppelin_speed_value = {

	value = 45 # Base, km/h

	#Engine

	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_3_var ?= flag:zeppelin_slot_3_basket_flag } } add = 0 } #Observation Basket

	#Slot 4 Utility
	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_4_var ?= flag:zeppelin_slot_4_airmail_flag } } add = -5 } #Airmail
	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_4_var ?= flag:zeppelin_slot_4_floats_flag } } add = -5 } #Floats
	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_4_var ?= flag:zeppelin_slot_4_cargo_flag } } add = -10 } #Cargo Hold

	#Slot 5 Extra Gondola
	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_lounge_flag } } add = 0 } #Lounge
	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_cage_flag } } add = 0 } #Ice Bear Cage
	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_carrier_flag } } add = -10 } #Airship Carrier

	if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_constructed_by_engineer_var ?= flag:zeppelin_constructed_by_engineer_flag } } multiply = 1.15 } #Engineer Bonus

	round = yes

#Zeppelin Construction Progress Value
zeppelin_construction_progress_value = {
	value = {
		desc = "tesla_engineering_progress_base_tt"
		value = 1000
	}
	#Having a Zeppelin Hangar
	progress_condition = {
		limit = {
			zeppelin_has_hangar_trigger = yes
		}
		add = {
			desc = "zeppelin_construction_progress_zeppelin_hangar_value_tt"
			value = 0.5
		}
	}
	#Having a Aeronautical Engineer
	progress_condition = {
		limit = {
			tesla_has_aeronautical_engineer_no_tooltip_trigger = yes
		}
		add = {
			desc = "tesla_engineering_progress_aeronautical_engineer_value_tt"
			value = 0.5
		}
	}
	#Engineer's Aeronautical Engineering Experience divided by 300
	progress_condition = {
		limit = {
			var:tesla_engineer_var ?= {
				has_variable = tesla_aeronautical_engineer_experience_var
				var:tesla_aeronautical_engineer_experience_var > 0
			}
		}
		add = {
			desc = "tesla_engineering_progress_aeronautical_experience_tt"
			value = tesla_engineering_progress_aeronautical_experience_value
		}
	}
	#Engineer has a Balloonist Trait
	progress_condition = {
		limit = {
			var:tesla_engineer_var ?= {
				OR = {
					has_trait = santos_balloonist
					has_trait = santos_balloonist_experienced
					has_trait = santos_balloonist_famous
				}
			}
		}
		add = {
			desc = "tesla_engineering_progress_balloonist_trait_tt"
			value = 0.5
		}
	}
}

		if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_4_var ?= flag:zeppelin_slot_4_airmail_flag } } add = 0 } #Airmail
		if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_4_var ?= flag:zeppelin_slot_4_floats_flag } } add = -500 } #Floats
		if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_4_var ?= flag:zeppelin_slot_4_cargo_flag } } add = -2500 } #Cargo Hold

		#Slot 5
		if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_lounge_flag } } add = 0 } #Lounge
		if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_cage_flag } } add = -1500 } #Ice Bear Cage
		if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_slot_5_var ?= flag:zeppelin_slot_5_carrier_flag } } add = -2000 } #Airship Carrier

		if = { limit = { var:zeppelin_captain_var ?= { var:zeppelin_constructed_by_engineer_var ?= flag:zeppelin_constructed_by_engineer_flag } } multiply = 1.15 } #Engineer Bonus

	}

	round = yes
}
