### Civil Engineering Projects ###

#Monthly Civil Engineering Progress
tesla_monthly_civil_engineering_progress_value = {
	value = {
		desc = "tesla_engineering_progress_base_tt"
		value = 1
	}

	#Having a Civil Engineer
	progress_condition = {
		limit = {
			tesla_has_civil_engineer_no_tooltip_trigger = yes
		}
		add = {
			desc = "tesla_engineering_progress_civil_engineer_value_tt"
			value = 1
		}
	}
	#Engineer's Civil Engineering Experience divided by 300
	progress_condition = {
		limit = {
			var:tesla_engineer_var ?= {
				has_variable = tesla_civil_engineer_experience_var
				var:tesla_civil_engineer_experience_var > 0
			}
		}
		add = {
			desc = "tesla_engineering_progress_civil_experience_value_tt"
			value = tesla_engineering_progress_civil_experience_value
		}
	}
	#Engineer has Civil Engineer Trait
	progress_condition = {
		limit = {
			var:tesla_engineer_var ?= {
				has_trait = tesla_civil_engineer
			}
		}
		add = {
			desc = "tesla_engineering_progress_civil_engineer_trait_tt"
			value = 1
		}
	}
	add = tesla_engineering_progress_engineer_with_meticulous_trait_value
	add = tesla_engineering_progress_engineer_with_persistent_trait_value
	add = tesla_engineering_progress_engineer_with_innovative_trait_value
	add = tesla_engineering_progress_engineer_with_surveyor_trait_value
	add = tesla_engineering_progress_engineer_with_physicist_trait_value
	add = tesla_engineering_progress_additional_support_value
	add = tesla_engineering_progress_mathematical_technical_universities_tiny_value
	add = mr_cheat_progress_value
}

#Civil Engineer Experience Value
tesla_engineering_progress_civil_experience_value = {
	value = var:tesla_engineer_var.var:tesla_civil_engineer_experience_var
	divide = 300
	max = 1
}

#Value for the Research Project
tesla_monthly_civil_engineering_progress_for_research_value = {
	value = 0
	#If Engineer is without project this project gets progress
	if = {
		limit = {
			tesla_has_active_engineer_no_tooltip_trigger = yes
		}
		add = tesla_monthly_civil_engineering_progress_value
	}
}
