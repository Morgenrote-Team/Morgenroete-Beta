#Mathematical Physicist Experience
curie_physics_progress_physicist_mathematical_physicist_experience_value = {
	progress_condition = {
		limit = {
			var:curie_physicist_var ?= {
				has_variable = curie_mathematical_physicist_experience_var
				var:curie_mathematical_physicist_experience_var > 0
			}
		}
		add = {
			desc = "curie_physics_project_progress_physics_experience_tt"
			value = var:curie_physicist_var.var:curie_mathematical_physicist_experience_var
			divide = 300
		}
	}
}
