#Yearly Physical Data Value
curie_yearly_physical_data_value = {
	value = 0
	###Technology###
	root = {
		if = {
			limit = {
					has_technology_researched = curie_mathematics_tech
			}
			add = 1
		}
		if = {
			limit = {
					has_technology_researched = verrier_spectroscopy_tech
			}
			add = 2
		}
		if = {
			limit = {
				has_technology_researched = verrier_radioactivity_tech
			}
			add = 3
		}
		if = {
			limit = {
				has_technology_researched = curie_nuclear_fission_tech
			}
			add = 4
		}
		###Observatory PMs###
		if = {
			limit = {
				any_scope_state = {
					is_production_method_active = {
						building_type = building_verrier_observatory
						production_method = pm_curie_magnetic_observatory
					}
				}
			}
			add = 1
		}
		###University with Mathematics Focus###
		if = {
			limit = {
				any_scope_building = {
					is_building_type = building_university
					has_active_production_method = pm_dubois_university_mathematics
				}
			}
			add = 2
		}
		###Physicist###
		if = {
			limit = {
				curie_has_physicist = yes
			}
			add = 3
		}
		#Meteorological Observation###
		if = {
			limit = {
				has_journal_entry = je_curie_meteorological_observation
				curie_has_active_meteorological_physicist_trigger = yes
			}
			add = curie_meteorological_observation_yearly_physical_data_value
		}
	}
}

#Value used to determine how many physical data the Meteorological Observation will produce per month
curie_meteorological_observation_yearly_physical_data_value = {
	#Base Physical Data
	value = 1

	#Observatory with Early Meteorology
	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_verrier_observatory
				has_active_production_method = pm_curtiss_early_weather_station
			}
		}
		add = 1
	}

	#Observatory with Standard Meteorology
	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_verrier_observatory
				has_active_production_method = pm_curtiss_weather_station
			}
		}
		add = 2
	}

	#Observatory with Modern Meteorology
	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_verrier_observatory
				has_active_production_method = pm_curtiss_modern_weather_station
			}
		}
		add = 3
	}

	#Experience of Meteorologist
	if = {
		limit = {
			curie_has_meteorological_physicist_trigger = yes
		}
		add = {
			value = var:curie_physicist_var.var:curie_meteorological_physicist_experience_var
			divide = 50
		}
	}

	round = yes
}

curie_meteorological_observation_yearly_physical_data_rounded_value = {
	value = curie_meteorological_observation_yearly_physical_data_value
	divide = 10
	round = yes
	multiply = 10
}
