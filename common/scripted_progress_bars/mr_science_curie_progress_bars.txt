curie_magnetic_station_progress_bar = {
	name = "curie_magnetic_station_progress_bar"
	desc = "curie_magnetic_station_progress_bar_desc"

	default_green = yes

	monthly_progress = {
		if = {
			limit = {
				has_variable = curie_physicist_var
			}
			add = {
				desc = "curie_physics_project_progress_base_value_tt"
				value = 1
			}
			if = {
				limit = {
					var:curie_physicist_var ?= {
						has_trait = curie_mathematical_physicist
					}
				}
				add = {
					desc = "curie_physics_project_mathematician_trait_tt"
					value = 1
				}
			}
			if = {
				limit = {
					var:curie_physicist_var ?= {
						has_trait = engineer
					}
				}
				add = {
					desc = "curie_physics_project_engineer_trait_tt"
					value = 0.5
				}
			}
			if = {
				limit = {
					var:curie_physicist_var ?= {
						OR = {
							has_trait = meticulous
							has_trait = persistent
						}
					}
				}
				add = {
					desc = "curie_physics_project_exact_trait_tt"
					value = 0.3
				}
			}
			if = {
				limit = {
					var:curie_physicist_var ?= {
						OR = {
							has_trait = alcoholic
							has_trait = opium_addiction
							has_trait = cocaine_addiction
							has_trait = syphilis
							has_trait = psychological_affliction
							has_trait = senile
							has_trait = panum_cholera
							has_trait = tuberculosis
							has_trait = panum_yellow_fever
						}
					}
				}
				add = {
					desc = "curie_physics_project_negative_body_trait_tt"
					value = -0.5
				}
			}
			#Specialized Universities
			if = {
				limit = {
					academics_mathematical_university_multiplier_tiny > 0
				}
				add = {
					desc = "mendelejew_chemistry_progress_mathematical_universities_tt"
					add = academics_mathematical_university_multiplier_tiny
				}
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 24
}
