### Balloon Ride ###

santos_balloon_ride_progress_bar = {
	name = "santos_balloon_ride_progress_bar"
	desc = "santos_balloon_ride_progress_bar_desc"

	default_green = yes

	monthly_progress = {
		if = {
			limit = {
				santos_has_active_balloonist_no_tooltip_trigger = yes
			}
			add = {
				desc = "santos_balloon_ride_progress_balloonist_tt"
				value = 1
			}
			if = {
				limit = {
					var:santos_balloonist_var ?= {
						has_variable = santos_balloonist_experience_var
					}
				}
				add = {
					desc = "santos_balloon_ride_progress_balloonist_experience_tt"
					value = var:santos_balloonist_var.var:santos_balloonist_experience_var
					divide = 300
				}
			}
			if = {
				limit = {
					var:santos_balloonist_var ?= {
						has_trait = charismatic
					}
				}
				add = {
					desc = "santos_balloon_ride_progress_balloonist_charismatic_trait_tt"
					value = 1
				}
			}
			if = {
				limit = {
					var:santos_balloonist_var ?= {
						has_trait = innovative
					}
				}
				add = {
					desc = "santos_balloon_ride_progress_balloonist_innovative_trait_tt"
					value = 0.5
				}
			}
			if = {
				limit = {
					var:santos_balloonist_var ?= {
						has_trait = reserved
					}
				}
				add = {
					desc = "santos_balloon_ride_progress_balloonist_reserved_trait_tt"
					value = -0.3
				}
			}
			if = {
				limit = {
					var:santos_balloonist_var ?= {
						OR = {
							has_trait = alcoholic
							has_trait = expensive_tastes
							has_trait = opium_addiction
							has_trait = cocaine_addiction
							has_trait = syphilis
							has_trait = psychological_affliction
							has_trait = senile
							has_trait = panum_cholera
							has_trait = tuberculosis
							has_trait = panum_yellow_fever
						}
					}
				}
				add = {
					desc = "santos_balloon_ride_progress_balloonist_negative_body_trait_tt"
					value = -0.5
				}
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 24
}
