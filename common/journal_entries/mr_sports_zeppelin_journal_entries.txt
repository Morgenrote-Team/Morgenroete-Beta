### Blau Gas ###

je_zeppelin_propellant_invention = {
	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	activation_description = {
		always = yes
	}

	group = je_group_curtiss_aviation

	scripted_progress_bar = zeppelin_propellant_invention_project_progress_bar

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			mendelejew_has_chemist = yes
			custom_tooltip = {
				text = zeppelin_propellant_invention_project_progress_bar_at_full_tt
				je:je_zeppelin_propellant_invention = {
					"scripted_bar_progress(zeppelin_propellant_invention_project_progress_bar)" >= 24
				}
			}
		}

		effect = {
			trigger_event = {
				id = zeppelin.111
			}
			mendelejew_set_chemist_unbusy_effect = yes
			var:mendelejew_chemist_var ?= {
				set_variable = {
					name = mendelejew_chemist_achievement_blau_gas_var
					value = flag:mendelejew_chemist_achievement_blau_gas_loc
				}
			}
		}
	}

	#Invalid
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_chemist_dies_tt
				NOT = {
					has_variable = mendelejew_chemist_var
				}
			}
		}
	}

	weight = 10
}

### Arctic Zeppelin Expedition ###

je_curtiss_zeppelin_expedition = {
	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	activation_description = {
		always = yes
	}

	group = je_group_curtiss_aviation

	current_value = {
		value = var:curtiss_zeppelin_progress
	}

	goal_add_value = {
		value = 16
	}

	progressbar = yes

	on_weekly_pulse = {
		random_events = {
			125 = curtiss.529 #Curtiss 529: Expedition progress
			5 = curtiss.527 #Curtiss 527: North Pole Passage
			20 = curtiss.528 #Curtiss 528: Rendez-Vous with $COUNTRY$
			10 = curtiss.531 #Curtiss 531: Discovery of ice free passage/Aurora Borealis/Icebears
			10 = curtiss.532 #Curtiss 532: New maps/Magnetic studies/Narwhal (storm may come)
			10 = curtiss.533 #Curtiss 533: Searching fossiles/Meteorological station/Surveying Arctica (land)
			10 = curtiss.534 #Curtiss 534: Strong winds
			10 = curtiss.535 #Curtiss 535: Storm
			10 = curtiss.536 #Curtiss 536: Clear weather
			10 = curtiss.537 #Curtiss 537: Icy propellers
			10 = curtiss.538 #Curtiss 538: Zeppelin anchorage
			10 = curtiss.539 #Curtiss 539: Engine failure!
			15 = curie.522 #Curie 522: Nacreous Cloud
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = curtiss_zeppelin_0_danger
				trigger = {
					var:curtiss_zeppelin_danger = 0
				}
			}
			triggered_desc = {
				desc = curtiss_zeppelin_1_danger
				trigger = {
					var:curtiss_zeppelin_danger = 1
				}
			}
			triggered_desc = {
				desc = curtiss_zeppelin_2_danger
				trigger = {
					var:curtiss_zeppelin_danger = 2
				}
			}
		}
	}

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
		}

		effect = {
			trigger_event = {
				id = curtiss.523
				popup = yes
			}
		}
	}

	#Failure
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"

		trigger = {
			custom_tooltip = {
				text = je_zeppelin_expedition_fail
				var:curtiss_zeppelin_danger >= 3
			}
		}

        effect = {
			trigger_event = {
				id = curtiss.524
				popup = yes
			}
        }
	}

	weight = 25

	should_be_pinned_by_default_uninvolved_or_context = yes
}
