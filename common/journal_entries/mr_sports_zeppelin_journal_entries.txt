### Zeppelin Construction ###
je_zeppelin_construction = {
	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	activation_description = {
		always = yes
	}

	group = je_group_curtiss_aviation

	scripted_button = zeppelin_stop_zeppelin_construction_button

	scripted_progress_bar = zeppelin_construction_progress_bar

	scaling_modifiers_while_active = {
		modifier = zeppelin_construction_cost_modifier
		value = zeppelin_construction_cost_value #Depends on size
	}

	immediate = {
		if = {
			limit = {
				zeppelin_captain_creation_trigger = yes
			}
			trigger_event = {
				id = zeppelin.101
			}
		}
	}

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			custom_tooltip = {
				text = je_zeppelin_construction_complete_tt
				zeppelin_construction_goal_trigger = yes
			}
		}

		effect = {
			zeppelin_construction_finished_effect = yes
			remove_variable = zeppelin_construction_target_var
		}
	}

	#Ended
	outcome = {
		header = mr_je_closing_conclusion_header
		color = "mr_je_color_grey"
		effect_header = mr_je_on_closing_conclusion_header
		conclusion_desc = mr_je_standard_closing_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = mr_je_closing_conclusion_trigger_tt
				always = no
			}
		}
		effect = {
			remove_variable = zeppelin_construction_target_var
			if = {
				limit = {
					zeppelin_has_captain = yes
				}
				retire_captain_effect = yes
			}
		}
	}

	weight = 100
}

je_zeppelin_propellant_invention = {
	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	activation_description = {
		always = yes
	}

	group = je_group_curtiss_aviation

	scripted_progress_bar = zeppelin_propellant_invention_project_progress_bar

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			mendelejew_has_chemist = yes
			custom_tooltip = {
				text = zeppelin_propellant_invention_project_progress_bar_at_full_tt
				je:je_zeppelin_propellant_invention = {
					"scripted_bar_progress(zeppelin_propellant_invention_project_progress_bar)" >= 24
				}
			}
		}

		effect = {
			trigger_event = {
				id = zeppelin.111
			}
			mendelejew_set_chemist_unbusy_effect = yes
			var:mendelejew_chemist_var ?= {
				set_variable = {
					name = mendelejew_chemist_achievement_blau_gas_var
					value = flag:mendelejew_chemist_achievement_blau_gas_loc
				}
			}
		}
	}

	#Invalid
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_chemist_dies_tt
				NOT = {
					has_variable = mendelejew_chemist_var
				}
			}
		}
	}

	weight = 10
}

### Arctic Zeppelin Expedition ###

je_zeppelin_expedition = {
	icon = "gfx/interface/icons/event_icons/event_zeppelin.dds"

	activation_description = {
		always = yes
	}

	group = je_group_curtiss_aviation

	scripted_progress_bar = zeppelin_expedition_progress_bar
	scripted_progress_bar = zeppelin_expedition_danger_bar

	widget = {
		gui = "gui/journal_entry_widgets/mr_character_je_widgets.gui"
		name = "zeppelin_expedition_header"
		container = "custom_widget_container_je_icon"
	}

	widget = {
		gui = "gui/journal_entry_widgets/zeppelin_je_widgets.gui"
		name = "widget_zeppelin_expedition_event_list_widget"
		container = "custom_widget_container_7"
	}

	modifiers_while_active = {
		zeppelin_expedition_cost_modifier
	}

	#Expedition Logs: Clear Weather (+1 Progress), Strong Winds (0 progress, +100 km/h), Icy Propellers (+1 Danger, Arctic Equipment), Superheat (+1 Danger, Synthetic-Latex), Engine failure (+1 Danger, High Performance Engine), Zeppelin Anchorage (-1 Danger)
	#Progress 2: Aurora Borealis/Ice Bear
	#Progress 5: First Arctic Region. Posibility to explore: Search Fossiles and Specimen (Ichthyostega/Meteorite/Storm), Build Meteorological Station (Nacreous Cloud)
	#Progress 8: Magnetic Studies/Narwhal
	#Progress 10: Second Arctic Region. No Possibility to explore.
	on_weekly_pulse = {
		events = {
			zeppelin.231 #Zeppelin 231 Aurora Borealis/Icebears
			zeppelin.232 #Zeppelin 232 Magnetic Studies/Narwhal
			zeppelin.233 #Zeppelin 233 Arctic Region
			zeppelin.234 #Zeppelin 234 Arctic Region
			zeppelin.235 #Zeppelin 235 North Pole Passage
		}
		effect = {
			if = {
				limit = {
					NOT = {
						has_variable = zeppelin_arctic_event_active_var
					}
				}
				trigger_event = {
					days = { 0 30 }
					on_action = zeppelin_expedition_on_action
				}
			}
		}
	}

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			custom_tooltip = {
				text = zeppelin_expedition_progress_bar_at_full_tt
				je:je_zeppelin_expedition = {
					"scripted_bar_progress(zeppelin_expedition_progress_bar)" >= zeppelin_expedition_goal_value
				}
			}
		}

		effect = {
			trigger_event = {
				id = zeppelin.202
			}
		}
	}

	#Failure
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"

		trigger = {
			custom_tooltip = {
				text = zeppelin_expedition_danger_bar_at_full_tt
				je:je_zeppelin_expedition = {
					"scripted_bar_progress(zeppelin_expedition_danger_bar)" >= zeppelin_expedition_danger_goal_value
				}
			}
		}

        effect = {
			trigger_event = {
				id = zeppelin.203
			}
        }
	}

	weight = 25

	should_be_pinned_by_default_uninvolved_or_context = yes
}
