###### GOETTINGER SEVEN MONUMENT #######################################################

je_goettinger_seven_monument = {
	icon = "gfx/interface/icons/event_icons/event_academy.dds"

	group = je_group_academics_academia

	can_deactivate = yes

	is_shown_when_inactive = {
		exists = c:GER
		c:GER = ROOT
		has_variable = academics_goettinger_seven_support_var
	}

	possible = {
		any_scope_state = {
			state_region = s:STATE_HANNOVER
		}
	}

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			ig:ig_intelligentsia = {
				is_in_government = yes
			}
			has_law = law_type:law_protected_speech
			OR = {
				has_law = law_type:law_wealth_voting
				has_law = law_type:law_census_voting
				has_law = law_type:law_universal_suffrage
			}
		}

		effect = {
			add_loyalists = {
				value = 0.15
				pop_type = academics
			}
			random_state = {
				limit = {
					state_region = s:STATE_HANNOVER
				}
				add_modifier = {
					name = academics_goettinger_seven_monument_modifier
				}
			}
		}
	}

	weight = 35
}
