je_curtiss_found_schneider_trophy_contextless = {
	icon = "gfx/interface/icons/event_icons/event_aviation.dds"

	group = je_group_vikelas_international_sports_events

	scripted_progress_bar = curtiss_nations_with_seaplanes_progress_bar

	should_be_involved = { always = no }

	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = {
					curtiss_schneider_trophy_possible = yes
					NOT = { is_involved_in_journal_entry = je_curtiss_found_schneider_trophy_contextless }
				}
				scope:journal_entry = {
					add_involved_country = prev
				}
			}
		}
	}

	complete = {
		scope:journal_entry = {
			custom_tooltip = {
				text = je_curtiss_found_schneider_trophy_contextless_seaplane_nations_number_trigger_tt
				any_scope_je_involved = {
					count >= 5
				}
			}
			custom_tooltip = {
				text = je_curtiss_found_schneider_trophy_contextless_seaplane_nations_founder_trigger_tt
				any_scope_je_involved = {
					country_rank >= rank_value:major_power
					any_scope_state = {
						is_incorporated = yes
						any_scope_building = {
							is_building_type = building_port
						}
						any_scope_building = {
							is_building_type = building_urban_center
							level >= 15
						}
					}
				}
			}
		}
	}

	on_complete = {
		scope:journal_entry = {
			random_scope_je_involved = {
				limit = {
					country_rank >= rank_value:major_power
					any_scope_state = {
						is_incorporated = yes
						any_scope_building = {
							is_building_type = building_port
						}
						any_scope_building = {
							is_building_type = building_urban_center
							level >= 15
						}
					}
				}
				trigger_event = {
					id = curtiss.100
					popup = yes
				}
			}
		}
	}

	# fires when a country becomes involved after activation of a contextless JE, ie not as a result of the should_be_involved trigger
	# root = specific country
	on_become_involved_after_activation = {
		scope:journal_entry = {
			add_progress = {
				name = curtiss_nations_with_seaplanes_progress_bar
				value = 1
			}
		}
	}

	# fires when a country stops being involved in a contextless JE
	# root = specific country
	on_no_longer_involved = {
		scope:journal_entry = {
			add_progress = {
				name = curtiss_nations_with_seaplanes_progress_bar
				value = -1
			}
		}
	}

	weight = 25
}

#Schneider Trophy JE
# This JE controlls the Trophy. Once a country wins 3 times in a row, the JE disappears
#
# The weekly pulse controls the Races themselves.
# The monthly pulse controls when the race takes place.
# The victory in a race is controlled in the schneider trophy points effect.
je_curtiss_schneider_trophy_race_contextless = {
	icon = "gfx/interface/icons/event_icons/event_aviation.dds"

	group = je_group_vikelas_international_sports_events

	#Add Scripted Button here

	should_be_involved = { always = no }

	widget = {
		gui = "gui/journal_entry_widgets/mr_character_je_widgets.gui"
		name = "curtiss_schneider_trophy_race_header"
		container = "custom_widget_container_je_icon"
	}

	widget = {
		gui = "gui/journal_entry_widgets/curtiss_je_widgets.gui"
		name = "widget_curtiss_schneider_trophy_widgets"
		container = "custom_widget_container_je_icon"
	}

	immediate = {
		scope:journal_entry = {
			set_variable = com_hide_journal_reason
			set_variable = com_hide_involved_countries
		}
	}

	#This on action controls the race itself. It only does anything if the race is currently ongoing.
	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				if = {
					limit = {
						has_variable = curtiss_schneider_trophy_race_active_var
					}
					#Race Events
					if = {
						limit = {
							has_variable = curtiss_schneider_trophy_race_active_var
						}
						every_scope_je_involved = {
							### Event Pulse
							trigger_event = {
								on_action = curtiss_schneider_trophy_events_on_action
							}
						}
					}
				}
			}
		}
	}

	#This on action controls when the race takes place and fires the events before the race itself starts.
	on_monthly_pulse = {
		effect = {
			scope:journal_entry = {
				#The first stage invites countries that are not currently part of the Schneider Trophy.
				if = {
					limit = {
						curtiss_schneider_trophy_time_till_next_race_value <= 180
						NOT = { has_variable = curtiss_schneider_trophy_preparation_stage_var }
					}
					#Starts preparation stages
					set_variable = {
						name = curtiss_schneider_trophy_preparation_stage_var
						value = 1
					}
					#Invites new Countries
					every_country = {
						limit = {
							curtiss_schneider_trophy_possible = yes
						}
						if = {
							limit = {
								NOT = { is_involved_in_journal_entry = je_curtiss_schneider_trophy_race_contextless }
							}
							trigger_event = {
								id = curtiss.101
								popup = yes
							}
						}
						else = {
							#Add a Notification
						}
					}
				}
				#The second stage fires preparation events (Financing, Pilot, etc.)
				else_if = {
					limit = {
						curtiss_schneider_trophy_time_till_next_race_value <= 120
						var:curtiss_schneider_trophy_preparation_stage_var ?= 1
					}
					#Sets next stage
					set_variable = {
						name = curtiss_schneider_trophy_preparation_stage_var
						value = 2
					}
					every_scope_je_involved = {
						set_variable = {
							name = schneider_trophy_point_var
							value = 0
						}
						#Curtiss 103: State Support
						trigger_event = {
							id = curtiss.103
							days = { 5 15 }
						}
						#Curtiss 108: Leading pilot
						trigger_event = {
							id = curtiss.108
							days = { 25 35 }
						}
						#Curtiss 140: [Aeronautical Engineer] constructs new engine
						trigger_event = {
							id = curtiss.140
							days = { 55 65 }
						}
					}
				}
				#The Race gets actually started
				else_if = {
					limit = { curtiss_schneider_trophy_time_till_next_race_value <= 10 }
					#Resets Stage
					remove_variable = curtiss_schneider_trophy_preparation_stage_var
					#Resets timer
					curtiss_schneider_trophy_set_time_till_next_race_effect = { DAYS = 730 }
					#Starts the Race
					set_variable = curtiss_schneider_trophy_race_active_var
					every_scope_je_involved = {
						#Sets Points up
						set_variable = {
							name = curtiss_schneider_trophy_point_var
							value = 0
						}
						curtiss_add_x_schneider_trophy_point_unscaled_effect = { AMOUNT = curtiss_schneider_trophy_before_race_points_score }
						#Removes Variables
						remove_variable = curtiss_schneider_trophy_training_good
						remove_variable = curtiss_schneider_trophy_training_normal
						remove_variable = curtiss_schneider_trophy_new_engine
					}
					curtiss_schneider_trophy_race_ranking_set_up_effect_effect = { JOURNAL_ENTRY_SCOPE = scope:journal_entry }
				}
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_curtiss_schneider_trophy_race_contextless_complete_tt
			scope:journal_entry = {
				has_variable = curtiss_end_of_schneider_trophy
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}
