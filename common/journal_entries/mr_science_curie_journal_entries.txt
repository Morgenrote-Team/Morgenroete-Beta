je_curie_magnetic_station = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	group = je_group_curie_physics

	scripted_progress_bar = curie_magnetic_station_progress_bar

	on_monthly_pulse = {
		random_events = {
			25 = 0
			1 = curie.104 #Gaudi 1404: Nautical interest
			1 = curie.105 #Gaudi 1405: An Anomaly
			1 = curie.106 #Gaudi 1406: Astronomer consults Geomagnetic Observatory
		}
	}

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			custom_tooltip = {
				text = je_curie_magnetic_station_progress_complete_tt
				scope:journal_entry = {
					"scripted_bar_progress(curie_magnetic_station_progress_bar)" >= 24
				}
			}
		}

		effect = {
			trigger_event = {
				id = curie.102
				popup = yes
			}

			#Starts new project for AI
			ai_start_physics_project_for_ai_effect = yes
		}
		curie_set_physicist_unbusy_effect = yes
	}

	#Invalid
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_physicist_dies_tt
				NOT = {
					has_variable = curie_physicist_var
				}
			}
		}

		effect = {
			curie_set_physicist_unbusy_effect = yes
		}
	}

	weight = 30

	should_be_pinned_by_default_uninvolved_or_context = yes
}

###Physics Experiment###
je_curie_physics_experiment = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	scripted_progress_bar = curie_physicist_experiment_project_progress_bar

	scripted_button = curie_stop_physicist_project_button

	group = je_group_curie_physics

	scaling_modifiers_while_active = {
		modifier = curie_physicist_experiment_cost_modifier
		value = mr_local_project_cost_very_small_value
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					NOR = {
						has_modifier = curie_physicist_experiment_progress_modifier
						has_modifier = curie_physicist_experiment_accident_modifier
						has_modifier = curie_physicist_experiment_disaster_modifier
					}
				}
				trigger_event = {
					id = curie.401
				}
			}
		}
	}

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			custom_tooltip = {
				text = je_curie_physics_experiment_complete
				scope:journal_entry = {
					"scripted_bar_progress(curie_physicist_experiment_project_progress_bar)" >= 40
				}
			}
		}

		effect = {
			if = {
				limit = {
					curie_experiment_can_discover_trigger = yes
				}
				custom_tooltip = {
					text = je_curie_physics_special_experiment_success
					curie_add_searched_experiment_from_list_effect = yes
					curie_set_physicist_unbusy_effect = yes
					trigger_event = {
						id = curie.402
					}
				}
			}
			else = {
				custom_tooltip = {
					text = je_curie_physics_generic_experiment_success
					curie_set_physicist_unbusy_effect = yes
					trigger_event = {
						id = curie.403
					}
				}
			}
		}
		curie_set_physicist_unbusy_effect = yes
	}

	#Invalid
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = curie_stop_physicist_project_tt
				OR = {
					has_variable = curie_stop_physics_project_var
					curie_has_physicist_no_tooltip_trigger = no
				}
			}
		}

        effect = {
			remove_variable = curie_stop_physics_project_var
			curie_set_physicist_unbusy_effect = yes
        }
	}

	#Failure
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"

		trigger = {
			custom_tooltip = {
				text = je_curie_physics_experiment_fail
				scope:journal_entry = {
					"scripted_bar_progress(curie_physicist_experiment_project_progress_bar)" <= -40
				}
			}
		}

        effect = {
			trigger_event = { id = curie.404 }
			curie_set_physicist_unbusy_effect = yes
        }
	}

	weight = 30
}

je_curie_improve_mathematics_project = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	scripted_progress_bar = curie_physicist_improve_mathematics_project_progress_bar

	scripted_button = curie_stop_physicist_project_button
	scripted_button = je_curie_improve_mathematics_project_change_list_button

	group = je_group_curie_physics

	scaling_modifiers_while_active = {
		modifier = curie_physicist_mathematics_cost_modifier
		value = mr_local_project_cost_very_small_value
	}

	immediate = {
		scope:journal_entry = {
			set_variable = {
				name = mr_num_improved_buildings_var
				value = 0
			}
		}
		ordered_in_list = {
			variable = curie_improve_mathematics_buildings_chosen_variable_list
			position = 0

			scope:journal_entry = {
				set_variable = {
					name = mr_target_building_var
					value = prev
				}
			}
		}
		scope:journal_entry = {
			mr_journal_entry_save_remaining_list_entries_as_variable_effect = { LIST_NAME = curie_improve_mathematics_buildings_chosen_variable_list }
		}
	}

	on_monthly_pulse = {
		effect = {
			mr_journal_entry_building_3_modifier_project_monthly_effect = {
				PROGRESS_BAR_NAME = curie_physicist_improve_mathematics_project_progress_bar
				PROGRESS_COST = 24
				MODIFIER_1 = curie_building_throughput_add_modifier
				MODIFIER_2 = curie_building_production_mult_modifier
				MODIFIER_3 = curie_building_goods_input_mult_modifier
				BUILDING_LIST_NAME = curie_improve_mathematics_buildings_chosen_variable_list
				ROLE_TYPE = academics
				ROLE = physicist
				ADDITIONAL_EFFECTS_BEFORE_BUILDING_REMOVAL_FROM_LIST = "curie_add_x_physical_data_effect = { AMOUNT = 1 }"
			}
		}
	}

	#Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"

		trigger = {
			custom_tooltip = {
				text = je_improve_building_project_complete_tt
				NOT = {
					has_variable_list = curie_improve_mathematics_buildings_chosen_variable_list
				}
			}
		}

		effect = {
			remove_variable = curie_stop_physics_project_var
			clear_variable_list = curie_improve_mathematics_buildings_chosen_variable_list

			curie_set_physicist_unbusy_effect = yes
			if = {
				limit = { curie_is_physicist_automated_trigger = no }
				trigger_event = {
					id = curie.301
				}
			}
			set_variable = {
				name = mr_num_improved_mathematics_buildings_var
				value = scope:journal_entry.var:mr_num_improved_buildings_var
			}
		}
	}

	#Invalid
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = curie_stop_physicist_project_tt
				OR = {
					has_variable = curie_stop_physics_project_var
					curie_has_physicist_no_tooltip_trigger = no
				}
			}
		}

        effect = {
			remove_variable = curie_stop_physics_project_var
			clear_variable_list = curie_improve_mathematics_buildings_chosen_variable_list
			curie_set_physicist_unbusy_effect = yes
        }
	}

	weight = 30
}

				name = mr_num_improved_meteorology_buildings_var
				value = scope:journal_entry.var:mr_num_improved_buildings_var
			}
		}
	}

	#Invalid
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = curie_stop_physicist_project_tt
				OR = {
					has_variable = curie_stop_physics_project_var
					curie_has_physicist_no_tooltip_trigger = no
				}
			}
		}

        effect = {
			remove_variable = curie_stop_physics_project_var
			clear_variable_list = curie_improve_meteorology_buildings_chosen_variable_list
			curie_set_physicist_unbusy_effect = yes
        }
	}

	weight = 30
}
