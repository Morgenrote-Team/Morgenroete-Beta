###Magnetic Association###

je_curie_magnetic_station = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	group = je_group_curie_physics

	scripted_progress_bar = curie_magnetic_station_progress_bar

	on_monthly_pulse = {
		random_events = {
			25 = 0
			1 = curie.104 #Gaudi 1404: Nautical interest
			1 = curie.105 #Gaudi 1405: An Anomaly
			1 = curie.106 #Gaudi 1406: Astronomer consults Geomagnetic Observatory
		}
	}

	immediate = {
	}

	complete = {
		custom_tooltip = {
			text = je_curie_magnetic_station_progress_complete_tt
			scope:journal_entry = {
				"scripted_bar_progress(curie_magnetic_station_progress_bar)" >= 24
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = curie.102
			popup = yes
		}

		#Starts new project for AI
		ai_start_physics_project_for_ai_effect = yes
	}

	invalid = {
		custom_tooltip = {
			text = je_physicist_dies_tt
			NOT = {
				has_variable = curie_physicist_var
			}
		}
	}

	on_invalid = {
	}

	weight = 30

	should_be_pinned_by_default_uninvolved_or_context = yes
}

###Discovery of a New Formula###
je_curie_formula_discovery = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	group = je_group_curie_physics

	scripted_progress_bar = curie_formula_discovery_progress_bar

	immediate = {
	}

	on_monthly_pulse = {
		random_events = {
			700 = 0

			#10 = mendelejew.211 #Mendelejew 211: A new Laboratory
			#10 = mendelejew.212 #Mendelejew 212: Unknown Chemical Compound
			#10 = mendelejew.213 #Mendelejew 213: Chemical Assistant
			#10 = mendelejew.214 #Mendelejew 214: Bad samples
			#10 = mendelejew.215 #Mendelejew 215: GEOLOGIST helps

			#20 = mendelejew.401 #Mendelejew 401: A Chemical Experiment
		}
	}

	complete = {
		custom_tooltip = {
			text = je_curie_magnetic_atlas_progress_complete_tt
			scope:journal_entry = {
				"scripted_bar_progress(curie_formula_discovery_progress_bar)" >= curie_searched_formula_duration_value
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = curie.201
		}
		curie_set_physicist_unbusy_effect = yes
	}

	invalid = {
		custom_tooltip = {
			text = je_physicist_dies_tt
			NOT = {
				has_variable = curie_physicist_var
			}
		}
	}

	on_invalid = {
		remove_variable = curie_searched_variable
	}

	fail = {
		custom_tooltip = {
			text = je_curie_formula_discovery_failed_tt
			any_in_global_list = {
				variable = curie_formulas_global_list
				var:curie_formula_number_var ?= prev.var:curie_searched_variable
				has_variable = curie_formula_discovered_var
			}
			scope:target = {
				NOT = { has_variable = curie_formula_discovered_var }
			}
		}
	}

	on_fail = {
		trigger_event = {
			id = curie.203
		}
		curie_set_physicist_unbusy_effect = yes
	}

	weight = 50
}
