je_curie_magnetic_station = {

			1 = curie.104 #Gaudi 1404: Nautical interest
			1 = curie.105 #Gaudi 1405: An Anomaly
			1 = curie.106 #Gaudi 1406: Astronomer consults Geomagnetic Observatory
		}
	}

	immediate = {
	}

	complete = {
		custom_tooltip = {
			text = je_curie_magnetic_station_progress_complete_tt
			scope:journal_entry = {
				"scripted_bar_progress(curie_magnetic_station_progress_bar)" >= 24
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = curie.102
			popup = yes
		}

		#Starts new project for AI
		ai_start_physics_project_for_ai_effect = yes
	}

	invalid = {
		custom_tooltip = {
			text = je_physicist_dies_tt
			NOT = {
				has_variable = curie_physicist_var
			}
		}
	}

	on_invalid = {
	}

	weight = 30

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_curie_mathematical_innovation = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	group = je_group_curie_physics

	immediate = {
		scope:journal_entry = {
			add_modifier = {
				name = curie_mathematical_innovation_modifier
				multiplier = curie_mathematical_innovation_value #Multiplies with the total innovation value
			}
		}
	}

	on_yearly_pulse = {
		effect = {
			scope:journal_entry = {
				remove_modifier = curie_mathematical_innovation_modifier
				add_modifier = {
					name = curie_mathematical_innovation_modifier
					multiplier = curie_mathematical_innovation_value #Multiplies with the total innovation value
				}
			}
		}
	}

	invalid = {
		custom_tooltip = {
			text = je_physicist_dies_tt
			NOT = {
				has_variable = curie_physicist_var
			}
		}
	}

	on_invalid = {
		scope:journal_entry = {
			remove_modifier = curie_mathematical_innovation_modifier
		}
	}

	weight = 50
}

###Discovery of a New Formula###
je_curie_formula_discovery = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	group = je_group_curie_physics

	scripted_progress_bar = curie_formula_discovery_progress_bar

	immediate = {
	}

	on_monthly_pulse = {
		random_events = {
			700 = 0

			#10 = mendelejew.211 #Mendelejew 211: A new Laboratory
			#10 = mendelejew.212 #Mendelejew 212: Unknown Chemical Compound
			#10 = mendelejew.213 #Mendelejew 213: Chemical Assistant
			#10 = mendelejew.214 #Mendelejew 214: Bad samples
			#10 = mendelejew.215 #Mendelejew 215: GEOLOGIST helps

			#20 = mendelejew.401 #Mendelejew 401: A Chemical Experiment
		}
	}

	complete = {
		custom_tooltip = {
			text = je_curie_magnetic_atlas_progress_complete_tt
			scope:journal_entry = {
				"scripted_bar_progress(curie_formula_discovery_progress_bar)" >= 60
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = curie.201
		}
		curie_set_physicist_unbusy_effect = yes
	}

	invalid = {
		custom_tooltip = {
			text = je_physicist_dies_tt
			NOT = {
				has_variable = curie_physicist_var
			}
		}
	}

	on_invalid = {
		remove_variable = curie_searched_variable
	}

	fail = {
		custom_tooltip = {
			text = je_curie_formula_discovery_failed_tt
			any_in_global_list = {
				variable = curie_formulas_global_list
				var:curie_formula_number_var ?= prev.var:curie_searched_variable
				has_variable = curie_formula_discovered_var
			}
			scope:target = {
				NOT = { has_variable = curie_formula_discovered_var }
			}
		}
	}

	on_fail = {
		trigger_event = {
			id = curie.203
		}
		curie_set_physicist_unbusy_effect = yes
	}

	weight = 50
}

###Improve Mathematics###
je_curie_improve_mathematics_project = {
	icon = "gfx/interface/icons/event_icons/event_physics.dds"

	scripted_progress_bar = curie_physicist_improve_mathematics_project_progress_bar

	scripted_button = curie_stop_physicist_project_button
	scripted_button = je_curie_improve_mathematics_project_change_list_button

	group = je_group_curie_physics

	immediate = {
		scope:journal_entry = {
			add_modifier = {
				name = curie_physicist_mathematics_cost_modifier
				multiplier = root.mr_local_project_cost_very_small_value
			}
			set_variable = {
				name = mr_num_improved_buildings_var
				value = 0
			}
		}
		ordered_in_list = {
			variable = curie_improve_mathematics_buildings_chosen_variable_list
			position = 0

			scope:journal_entry = {
				set_variable = {
					name = mr_target_building_var
					value = prev
				}
			}
		}
		scope:journal_entry = {
			mr_journal_entry_save_remaining_list_entries_as_variable_effect = { LIST_NAME = curie_improve_mathematics_buildings_chosen_variable_list }
		}
	}

	on_monthly_pulse = {
		effect = {
			mr_journal_entry_building_3_modifier_project_monthly_effect = {
				PROGRESS_BAR_NAME = curie_physicist_improve_mathematics_project_progress_bar
				PROGRESS_COST = 24
				MODIFIER_1 = curie_building_throughput_add_modifier
				MODIFIER_2 = curie_building_production_mult_modifier
				MODIFIER_3 = curie_building_goods_input_mult_modifier
				BUILDING_LIST_NAME = curie_improve_mathematics_buildings_chosen_variable_list
				ROLE_TYPE = academics
				ROLE = physicist
				ADDITIONAL_EFFECTS_BEFORE_BUILDING_REMOVAL_FROM_LIST = "curie_add_x_physical_data_effect = { AMOUNT = 1 }"
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_improve_building_project_complete_tt
			NOT = {
				has_variable_list = curie_improve_mathematics_buildings_chosen_variable_list
			}
		}
	}

	on_complete = {
		remove_variable = curie_stop_physicist_improvement_project_var
		show_as_tooltip = {
			scope:journal_entry = {
				remove_modifier = curie_physicist_mathematics_cost_modifier
			}
		}
		clear_variable_list = curie_improve_mathematics_buildings_chosen_variable_list

		curie_set_physicist_unbusy_effect = yes
		if = {
			limit = { ito_is_naturalist_automated_trigger = no }
			trigger_event = {
				id = curie.51
				popup = yes
			}
		}
		set_variable = {
			name = mr_num_improved_mathematics_buildings_var
			value = scope:journal_entry.var:mr_num_improved_buildings_var
		}
	}

	invalid = {
		custom_tooltip = {
			text = curie_stop_physicist_improvement_project_tt
			OR = {
				has_variable = curie_stop_physicist_improvement_project_var
				curie_has_physicist_no_tooltip_trigger = no
			}
		}
	}

	on_invalid = {
		remove_variable = curie_stop_physicist_improvement_project_var
		show_as_tooltip = {
			scope:journal_entry = {
				remove_modifier = curie_physicist_mathematics_cost_modifier
			}
		}
		clear_variable_list = curie_improve_mathematics_buildings_chosen_variable_list
		curie_set_physicist_unbusy_effect = yes
	}

	weight = 30
}
