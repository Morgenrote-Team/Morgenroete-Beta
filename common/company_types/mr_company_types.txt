		any_scope_building = {

				is_building_type = building_tesla_telegraph
				level > 0
			}

	prosperity_modifier = {

		building_electrics_industry_throughput_add = 0.05
		country_tech_spread_mult = 0.15
	}

	ai_will_do = {
		has_technology_researched = electric_telegraph
		any_scope_building = {
			filter = {
				is_building_type = building_tesla_telegraph
				level > 0
			}
			percent >= 0.65
		}
	}

	ai_weight = {
		value = 0
		if = {
			limit = {
				country_rank >= rank_value:great_power
			}
			add = 2
		}
		if = {
			limit = {
				exists = c:GBR
				c:GBR = ROOT
			}
			add = 1
		}
		if = {
			limit = {
				exists = c:USA
				c:USA = ROOT
			}
			add = 1
		}
		if = {
			limit = {
				exists = c:FRA
				c:USA = ROOT
			}
			add = 1
		}
	}
